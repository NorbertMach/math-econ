\chapter{Unconstrained Optimization in Several Variables}

When a firm produces two goods or hires two factors, profit and cost depend
on a vector of choice variables. The optimum is no longer characterised by a
single equation but by a \emph{system}: the gradient must vanish, and the
Hessian must have the right \emph{definiteness}. Sylvester's criterion
translates this into easy sign checks on a $2 \times 2$ matrix, and
convexity (or concavity) of the objective on the entire domain promotes a
local result to a global one. The drill below runs from pure classification
exercises to the standard economic applications: profit maximisation with two inputs
or two products, cost minimisation, and a monopolist pricing two
interdependent goods.

\section{Core Rules}

\begin{rulebox}{Gradient and the first-order condition}
Let $f : \R^n \to \R$ be $C^1$. A point $\mathbf{x}^* \in \R^n$ is a
\emph{critical point} of $f$ if
\[
\grad f(\mathbf{x}^*) \;=\; \mathbf{0}, \qquad \text{i.e.\ }
\pd{f}{x_i}(\mathbf{x}^*) = 0 \text{ for } i = 1, \dots, n.
\]
This is the necessary FOC for an interior extremum.
\end{rulebox}

\begin{rulebox}{Hessian and the second-order condition}
Assume $f$ is $C^2$ and let $H_f(\mathbf{x}^*)$ denote the Hessian at
$\mathbf{x}^*$. The SOC reads:
\begin{itemize}
    \item $H_f(\mathbf{x}^*)$ \textbf{negative definite (ND)} $\Rightarrow$ strict local \textbf{maximum};
    \item $H_f(\mathbf{x}^*)$ \textbf{positive definite (PD)} $\Rightarrow$ strict local \textbf{minimum};
    \item $H_f(\mathbf{x}^*)$ \textbf{indefinite} $\Rightarrow$ \textbf{saddle point};
    \item $H_f(\mathbf{x}^*)$ singular (det $=0$) $\Rightarrow$ inconclusive.
\end{itemize}
\end{rulebox}

\begin{rulebox}{The necessary side: semidefinite Hessians}
The box above lists \emph{sufficient} conditions. The converse direction gives only weaker,
\emph{necessary} ones (definite becomes semidefinite). Let $f$ be $C^2$ and $\mathbf{x}^*$ an interior
point.
\begin{itemize}[leftmargin=1.6em,itemsep=0.3ex]
  \item Local \textbf{minimum} at $\mathbf{x}^*$ $\Rightarrow$
        $\grad f(\mathbf{x}^*)=\mathbf{0}$ and $H_f(\mathbf{x}^*)$
        \textbf{positive semidefinite}. Local \textbf{maximum}
        $\Rightarrow$ gradient zero and $H_f(\mathbf{x}^*)$
        \textbf{negative semidefinite}. Reason: along every line
        $\varphi(t)=f(\mathbf{x}^*+t\mathbf{h})$ has a local minimum at
        $t=0$, so $\varphi''(0)=\mathbf{h}^{T}H_f(\mathbf{x}^*)\mathbf{h}\ge0$.
  \item Only semidefinite, not definite: $x^{4}+y^{4}$ has a strict
        minimum at the origin with $H=0$.
  \item Contrapositive: a Hessian that is not PSD rules out a minimum, one
        that is not NSD rules out a maximum, and an indefinite one rules out
        both, which is why it proves a saddle.
  \item A \textbf{semidefinite but singular} Hessian at a critical point
        decides nothing by itself. $x^{2}+y^{4}$ (minimum) and $x^{2}-y^{4}$
        (saddle) both have $H(0,0)=\operatorname{diag}(2,0)$, PSD and
        singular. All it does is rule out the opposite extremum unless
        $H=0$: a PSD $H\neq0$ excludes a maximum. Settle the case with the
        box on singular Hessians below.
\end{itemize}
Sufficient (definite) and necessary (semidefinite) differ exactly by the
singular case; \chref{derivatives}{11} draws the same line.
\end{rulebox}

\begin{rulebox}{Sylvester's criterion ($2\times2$)}
For
\[
H \;=\; \begin{pmatrix} a & b \\ b & c \end{pmatrix},
\qquad D_1 = a, \qquad D_2 = \det H = a c - b^2,
\]
the rules are
\[
\begin{array}{ll}
D_1 > 0,\; D_2 > 0 & \Rightarrow H \text{ PD (local min)}, \\
D_1 < 0,\; D_2 > 0 & \Rightarrow H \text{ ND (local max)}, \\
D_2 < 0 & \Rightarrow H \text{ indefinite (saddle)}, \\
D_2 = 0 & \Rightarrow \text{inconclusive}.
\end{array}
\]
Equivalently, the eigenvalues of $H$ have signs $(+,+)$, $(-,-)$, or
$(+,-)$.
\end{rulebox}

\begin{rulebox}{Leading principal minors in $n$ variables}
For $f : \R^n \to \R$ let $D_1, D_2, \dots, D_n$ be the leading principal
minors of $H_f(\mathbf{x}^*)$. Then $H_f(\mathbf{x}^*)$ is PD iff
$D_k > 0$ for all $k = 1, \dots, n$, and ND iff the signs alternate
starting negative: $D_1 < 0$, $D_2 > 0$, $D_3 < 0$, \dots
\begin{itemize}[leftmargin=1.6em,itemsep=0.3ex]
  \item If all $D_k\neq0$ but the signs fit \emph{neither} pattern, $H$ is
        \textbf{indefinite} and $\mathbf{x}^*$ is a \textbf{saddle point}.
        (In particular $D_{2k}<0$ for some $k$ always means a saddle.)
  \item If some $D_k=0$ with $k<n$ but $\det H\neq0$, $H$ cannot be
        definite (that needs every $D_k\neq0$), and a non-singular matrix
        that is not definite is indefinite: again a \textbf{saddle}. A
        negative $2\times2$ principal minor, or two diagonal entries of
        opposite sign, shows the same at a glance.
  \item If $\det H=0$, the test is silent: argue directly (box below).
\end{itemize}
\end{rulebox}

\begin{rulebox}{Global optima via concavity and convexity}
If $H_f$ is ND \emph{everywhere}, $f$ is strictly concave on $\R^n$ and any
critical point is the unique global maximum. Analogously for PD and minima.
This is why a local check delivers a global verdict for concave profit and
convex cost specifications.
\end{rulebox}

\begin{rulebox}{Singular Hessian: compare function values}
When $\det H_f(\mathbf{x}^*)=0$ the second-order test is silent and the
point must be classified directly, by comparing $f$ near $\mathbf{x}^*$
with $f(\mathbf{x}^*)$.
\begin{itemize}[leftmargin=1.6em,itemsep=0.3ex]
  \item \textbf{Restrict to lines or curves.} Study
        $\varphi(t)=f(\mathbf{x}^*+t\mathbf{h})$, especially along a null
        direction $\mathbf{h}$ of the Hessian ($H\mathbf{h}=\mathbf{0}$).
        If some restriction takes values both above and below
        $f(\mathbf{x}^*)$ arbitrarily close to $t=0$ (a $t^{3}$-type term),
        $\mathbf{x}^*$ is a \textbf{saddle point}. If two lines give a
        strict local minimum and a strict local maximum, it is a saddle as
        well.
  \item \textbf{Sign of $f-f(\mathbf{x}^*)$.} If
        $f(\mathbf{x})-f(\mathbf{x}^*)\ge0$ near $\mathbf{x}^*$, e.g.\ a sum
        of even powers, the point is a local minimum (strict if equality
        holds only at $\mathbf{x}^*$); $\le0$ gives a local maximum.
  \item \textbf{Non-isolated critical points.} The FOC may hold on a whole
        line or curve. Classify each point separately by the sign of
        $f-f(\mathbf{x}^*)$ on a small neighbourhood; such extrema are never
        strict.
\end{itemize}
A minimum along every line through $\mathbf{x}^*$ does \emph{not} by itself
prove a local minimum in general, so prefer the sign argument whenever $f$
can be written as a sum of signed terms.
\end{rulebox}

\begin{rulebox}{Quadratic functions in matrix form}
Let $A$ be a symmetric $n\times n$ matrix, $\mathbf{b}\in\R^{n}$, $c\in\R$,
and $f(\mathbf{x})=\mathbf{x}^{T}A\mathbf{x}+\mathbf{b}^{T}\mathbf{x}+c$.
In sum notation $f=\sum_{i}\sum_{j}a_{ij}x_ix_j+\sum_ib_ix_i+c$, and
differentiating with respect to $x_k$ hits the double sum once as $x_i$ and
once as $x_j$:
\[
  \pd{f}{x_k}=\sum_{j}a_{kj}x_j+\sum_{i}a_{ik}x_i+b_k
  =2(A\mathbf{x})_k+b_k ,
\]
using $a_{ik}=a_{ki}$. Hence
\[
  \grad f(\mathbf{x})=2A\mathbf{x}+\mathbf{b},\qquad H_f=2A\ \text{(constant)},
  \qquad
  \mathbf{x}^*=-\tfrac12A^{-1}\mathbf{b},\qquad
  f(\mathbf{x}^*)=c-\tfrac14\,\mathbf{b}^{T}A^{-1}\mathbf{b}
\]
when $A$ is invertible. The definiteness of $A$ classifies $\mathbf{x}^*$
\emph{globally}: PD, unique global minimum; ND, unique global maximum;
indefinite, saddle point. If $A$ is singular there is either no critical
point or a whole line (plane) of them. Remember the halving: a term
$\gamma x_ix_j$ ($i\neq j$) puts $\gamma/2$ into both $a_{ij}$ and $a_{ji}$.
\end{rulebox}

\begin{rulebox}{Least squares: the regression line by calculus}
Given data $(x_1,y_1),\dots,(x_n,y_n)$, not all $x_i$ equal, the line
$y=b_0+b_1x$ minimises the sum of squared residuals
\[
  S(b_0,b_1)=\sum_{i=1}^{n}\bigl(y_i-b_0-b_1x_i\bigr)^{2},
\]
a function of the two unknowns $b_0,b_1$; the data are constants.
Setting $\pd{S}{b_0}=-2\sum(y_i-b_0-b_1x_i)=0$ and
$\pd{S}{b_1}=-2\sum x_i(y_i-b_0-b_1x_i)=0$ gives the \emph{normal equations}
\[
  n\,b_0+\Bigl(\sum x_i\Bigr)b_1=\sum y_i,\qquad
  \Bigl(\sum x_i\Bigr)b_0+\Bigl(\sum x_i^{2}\Bigr)b_1=\sum x_iy_i ,
\]
with solution
\[
  b_1=\frac{S_{xy}}{S_{xx}},\qquad b_0=\bar y-b_1\bar x,\qquad
  S_{xx}=\sum(x_i-\bar x)^{2},\quad S_{xy}=\sum(x_i-\bar x)(y_i-\bar y).
\]
The Hessian is constant,
$H_S=2\begin{pmatrix}n&\sum x_i\\\sum x_i&\sum x_i^{2}\end{pmatrix}$, with
$D_1=2n>0$ and $D_2=4\bigl(n\sum x_i^{2}-(\sum x_i)^{2}\bigr)=4nS_{xx}>0$:
$S$ is strictly convex and the critical point is the unique global minimum.
In matrix form ($X$ with rows $(1,x_i)$) the normal equations read
$X^{T}X\,\mathbf{b}=X^{T}\mathbf{y}$, so $\mathbf{b}=(X^{T}X)^{-1}X^{T}\mathbf{y}$,
the formula of \chref{matrices}{10}, and $H_S=2X^{T}X$. Residuals are squared so that
positive and negative errors cannot cancel, large errors weigh more, and $S$
is differentiable (absolute values would leave kinks).
\end{rulebox}

\section{Worked Example}

\begin{worked}{A two-product monopolist: maximise $\pi(q_{1},q_{2})=90q_{1}-2q_{1}^{2}+60q_{2}-q_{2}^{2}-q_{1}q_{2}$}
\wstep{Step 1. First-order conditions: the gradient vanishes.}
\[
  \pd{\pi}{q_{1}}=90-4q_{1}-q_{2}=0,
  \qquad
  \pd{\pi}{q_{2}}=60-2q_{2}-q_{1}=0 .
\]
Note that each condition involves \emph{both} variables --- the cross term
$-q_{1}q_{2}$ links the two markets, so they cannot be optimised separately.
\wtrap{With a cross term present, maximising over $q_{1}$ with $q_{2}$ held
at some guess, and then over $q_{2}$, does not converge to the optimum in
one pass. The two conditions form a \emph{system} and must be solved
simultaneously.}

\wstep{Step 2. Solve the system.}
The second equation gives $q_{1}=60-2q_{2}$. Substituting into the first:
\[
  90-4(60-2q_{2})-q_{2}=0
  \;\Longrightarrow\;
  90-240+8q_{2}-q_{2}=0
  \;\Longrightarrow\;
  7q_{2}=150 ,
\]
\[
  q_{2}^{*}=\tfrac{150}{7}\approx21.43,
  \qquad
  q_{1}^{*}=60-\tfrac{300}{7}=\tfrac{120}{7}\approx17.14 .
\]
Check in the first condition: $90-4(120/7)-150/7=90-480/7-150/7
=90-630/7=90-90=0$. \checkmark

\wstep{Step 3. Second-order condition: build the Hessian.}
\[
  \pi_{11}=-4,\qquad \pi_{22}=-2,\qquad \pi_{12}=\pi_{21}=-1,
  \qquad
  \Hess\pi=\begin{pmatrix}-4&-1\\-1&-2\end{pmatrix}.
\]
\[
  D_{1}=-4<0,
  \qquad
  D_{2}=\det=8-1=7>0 .
\]
Alternating signs starting negative: \textbf{negative definite}, so the
critical point is a strict local maximum.
\wtrap{For a \emph{maximum} the leading principal minors must alternate
$-,+,-,\dots$ For a minimum they must all be positive. Requiring ``all
minors negative'' is the standard misremembering, and it rejects every
genuine maximum.}

\wstep{Step 4. Upgrade to a global statement.}
The Hessian is \emph{constant} --- $\pi$ is a quadratic --- so it is
negative definite at every point, meaning $\pi$ is strictly concave on all
of $\R^{2}$. A strictly concave function has at most one critical point and
it is the global maximum:
\[
  \boxed{(q_{1}^{*},q_{2}^{*})=\Bigl(\tfrac{120}{7},\tfrac{150}{7}\Bigr),
         \qquad \pi^{*}=\tfrac{69\,300}{49}\approx1414.3 .}
\]
\wtrap{This upgrade is available only because the Hessian is negative
definite \emph{everywhere}, not merely at the critical point. For a
non-quadratic objective, definiteness at one point gives a local result and
nothing more.}

\wstep{Step 5. Sanity-check the answer against the economics.}
Both quantities are positive, so the implicit constraints $q_{i}\ge0$ are
slack. \checkmark\ The cross term is negative, so the goods are
substitutes in production or demand; that is why each optimal quantity is
lower than it would be if the other market did not exist. Setting $q_{2}=0$
in the first condition would give $q_{1}=22.5$, above the joint optimum
$17.14$ --- exactly the interaction the model was built to capture.

\wstep{Step 6. The procedure, and the shape of the failure modes.}
\begin{center}\footnotesize
\begin{tabular}{ll}
  \hline
  step & what can go wrong\\
  \hline
  1. solve $\grad\pi=\mathbf 0$ & solving the equations one at a time\\
  2. build $\Hess\pi$ at each solution & mis-signed cross partials\\
  3. classify by minors & wrong sign pattern for a maximum\\
  4. check non-negativity & an ``optimum'' with $q_{i}<0$\\
  5. global? & claiming global from a local test\\
  \hline
\end{tabular}
\end{center}

\wstep{Step 7. What happens with three or more variables.}
The pattern extends: $\grad f=\mathbf 0$ is $n$ equations in $n$ unknowns,
and the classification uses the $n$ leading principal minors
$D_{1},\dots,D_{n}$ --- all positive for a minimum, alternating
$-,+,-,\dots$ for a maximum, and any other pattern with all $D_{i}\neq0$
means a saddle. When some $D_{i}=0$ the test is silent and you must argue
directly, as in \chref{derivatives}{11}.
\end{worked}

\section{Practice Problems}

For each function, locate all critical points and classify them using the
Hessian.

\begin{problems}
    \item $f(x, y) = x^2 + y^2 - 4 x + 6 y + 13$.
    \item $f(x, y) = -x^2 - 2 y^2 + 6 x + 8 y$.
    \item $f(x, y) = x y$.
    \item $f(x, y) = x^2 - y^2$.
    \item $f(x, y) = x^2 + x y + y^2 - 3 x$.
    \item $f(x, y) = x^3 - 3 x y + y^3$.
    \item $f(x, y) = x^2 + y^2 + x y - x - 2 y$.
    \item A two-product monopolist faces inverse demands $p_1 = 20 - q_1$ and $p_2 = 30 - 2 q_2$, with total cost $C(q_1, q_2) = q_1^2 + q_2^2 + q_1 q_2$. Find profit-maximising $q_1^*, q_2^*$ and verify the SOC.
    \item Minimise the cost $C(K, L) = 4 K^2 + 3 L^2 - 2 K L - 8 K - 6 L + 20$.
    \item $f(x, y) = \mathrm{e}^{x^2 + y^2}$ (locate the critical point and classify).
    \item A firm uses two inputs $x, y$ to produce output $Q = 10 x + 14 y - x^2 - y^2 - x y$. Find the input combination maximising $Q$ (interpret as profit with normalised prices).
    \item $f(x, y) = x^3 + y^3 - 3 x - 12 y + 7$. Locate and classify all critical points.
    \item $f(x, y) = x^2 y - x y^2 + x y$. Find all critical points and classify each. Does any of them have $x, y > 0$?
    \item A monopolist sells two substitute goods with demands $q_1 = 12 - 2 p_1 + p_2$ and $q_2 = 18 + p_1 - 2 p_2$ (so revenues depend on chosen prices). Marginal cost is zero. Maximise total revenue over $(p_1, p_2)$.
    \item Minimise $f(x, y) = x^2 + 4 y^2 + 2 x y - 6 x + 4$.
    \item Show $f(x,y) = x^2 + y^2 + xy$ is strictly convex; find its unique minimum.
    \item For $f(x,y,z) = x^2 + y^2 + z^2 - 2x - 4y - 6z$, find the global minimum.
    \item Decide if $f(x,y) = x^4 + y^4$ has a strict local minimum at $(0,0)$ via Hessian, and verify by another argument.
    \item Prove that $f(x,y) = \mathrm{e}^x + \mathrm{e}^y - x - y$ has a unique global minimum and find it.
    \item For what values of $a$ is $(0,0)$ a strict local minimum of $f(x,y) = x^2 + axy + y^2$?
\end{problems}

\section{Solutions}

\begin{solutions}
    \item $f_x = 2 x - 4$, $f_y = 2 y + 6$. FOC: $x = 2$, $y = -3$. $H = \begin{pmatrix} 2 & 0 \\ 0 & 2 \end{pmatrix}$, $D_1 = 2, D_2 = 4$, PD $\Rightarrow$ \textbf{min}. $f(2, -3) = 4 + 9 - 8 - 18 + 13 = 0$.
    \item $f_x = -2 x + 6 = 0 \Rightarrow x = 3$; $f_y = -4 y + 8 = 0 \Rightarrow y = 2$. $H = \begin{pmatrix} -2 & 0 \\ 0 & -4 \end{pmatrix}$, $D_1 = -2, D_2 = 8$, ND $\Rightarrow$ \textbf{max}. $f(3, 2) = -9 - 8 + 18 + 16 = 17$.
    \item $f_x = y = 0$, $f_y = x = 0 \Rightarrow (0, 0)$. $H = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}$, $D_2 = -1 < 0$, indefinite $\Rightarrow$ \textbf{saddle}.
    \item $f_x = 2 x = 0$, $f_y = -2 y = 0 \Rightarrow (0, 0)$. $H = \begin{pmatrix} 2 & 0 \\ 0 & -2 \end{pmatrix}$, $D_2 = -4 < 0$, indefinite $\Rightarrow$ \textbf{saddle}. Classical example.
    \item $f_x = 2 x + y - 3 = 0$, $f_y = x + 2 y = 0$. From the 2nd, $x = -2 y$. Sub: $-4 y + y - 3 = 0 \Rightarrow y = -1$, $x = 2$. $H = \begin{pmatrix} 2 & 1 \\ 1 & 2 \end{pmatrix}$, $D_1 = 2, D_2 = 3$, PD $\Rightarrow$ \textbf{min}.
    \item $f_x = 3 x^2 - 3 y = 0 \Rightarrow y = x^2$. $f_y = -3 x + 3 y^2 = 0 \Rightarrow x = y^2$. Sub: $x = (x^2)^2 = x^4 \Rightarrow x(x^3 - 1) = 0 \Rightarrow x = 0$ or $x = 1$. Critical points: $(0, 0)$ and $(1, 1)$. $H = \begin{pmatrix} 6 x & -3 \\ -3 & 6 y \end{pmatrix}$. At $(0,0)$: $\det = -9 < 0$, \textbf{saddle}. At $(1, 1)$: $D_1 = 6 > 0$, $D_2 = 36 - 9 = 27 > 0$, PD $\Rightarrow$ \textbf{min}.
    \item $f_x = 2 x + y - 1 = 0$, $f_y = 2 y + x - 2 = 0$. Solve: from (1), $y = 1 - 2 x$. Sub: $2(1 - 2 x) + x - 2 = -3 x = 0 \Rightarrow x = 0$, $y = 1$. $H = \begin{pmatrix} 2 & 1 \\ 1 & 2 \end{pmatrix}$, $D_2 = 3 > 0$, PD $\Rightarrow$ \textbf{min}.
    \item Revenue $R = p_1 q_1 + p_2 q_2 = (20 - q_1) q_1 + (30 - 2 q_2) q_2 = 20 q_1 - q_1^2 + 30 q_2 - 2 q_2^2$. Profit
    \[\pi = 20 q_1 - q_1^2 + 30 q_2 - 2 q_2^2 - q_1^2 - q_2^2 - q_1 q_2 = 20 q_1 + 30 q_2 - 2 q_1^2 - 3 q_2^2 - q_1 q_2.\]
    FOC: $\pi_{q_1} = 20 - 4 q_1 - q_2 = 0$; $\pi_{q_2} = 30 - 6 q_2 - q_1 = 0$. From the first, $q_2 = 20 - 4 q_1$. Sub: $30 - 6(20 - 4 q_1) - q_1 = 30 - 120 + 24 q_1 - q_1 = -90 + 23 q_1 = 0 \Rightarrow q_1^* = 90/23$, $q_2^* = 20 - 360/23 = (460 - 360)/23 = 100/23$.
    $H = \begin{pmatrix} -4 & -1 \\ -1 & -6 \end{pmatrix}$, $D_1 = -4 < 0$, $D_2 = 24 - 1 = 23 > 0$, ND $\Rightarrow$ \textbf{max}.
    \item $C_K = 8 K - 2 L - 8 = 0$, $C_L = 6 L - 2 K - 6 = 0$. From (1): $K = 1 + L/4$. Sub into (2): $6 L - 2(1 + L/4) - 6 = 6 L - 2 - L/2 - 6 = (11/2) L - 8 = 0 \Rightarrow L^* = 16/11$. Then $K^* = 1 + 4/11 = 15/11$. $H = \begin{pmatrix} 8 & -2 \\ -2 & 6 \end{pmatrix}$, $D_1 = 8 > 0$, $D_2 = 48 - 4 = 44 > 0$, PD $\Rightarrow$ \textbf{min}.
    \item $f_x = 2 x \, \mathrm{e}^{x^2 + y^2}$, $f_y = 2 y \, \mathrm{e}^{x^2 + y^2}$. The exponential is positive, so FOC gives $(x, y) = (0, 0)$. At the origin, $f_{xx} = 2 \mathrm{e}^0 + (2 x)^2 \mathrm{e}^0 = 2$, $f_{yy} = 2$, $f_{xy} = (2 x)(2 y) \mathrm{e}^0 = 0$. $H = 2 I$, PD $\Rightarrow$ \textbf{min}. (Indeed $f \geq 1$ with equality at $(0,0)$.)
    \item $Q_x = 10 - 2 x - y = 0$, $Q_y = 14 - 2 y - x = 0$. From the first: $y = 10 - 2 x$. Sub: $14 - 2(10 - 2 x) - x = 14 - 20 + 4 x - x = -6 + 3 x = 0 \Rightarrow x^* = 2$, $y^* = 6$. $H = \begin{pmatrix} -2 & -1 \\ -1 & -2 \end{pmatrix}$, $D_1 = -2, D_2 = 4 - 1 = 3 > 0$, ND $\Rightarrow$ \textbf{max}. $Q^* = 20 + 84 - 4 - 36 - 12 = 52$.
    \item $f_x = 3 x^2 - 3 = 0 \Rightarrow x = \pm 1$; $f_y = 3 y^2 - 12 = 0 \Rightarrow y = \pm 2$. Four critical points. $H = \begin{pmatrix} 6 x & 0 \\ 0 & 6 y \end{pmatrix}$, $D_2 = 36 x y$, $D_1 = 6 x$.
    \begin{itemize}
        \item $(1, 2)$: $D_1 = 6, D_2 = 72$, PD $\Rightarrow$ \textbf{min}.
        \item $(1, -2)$: $D_2 = -72 < 0$, \textbf{saddle}.
        \item $(-1, 2)$: $D_2 = -72 < 0$, \textbf{saddle}.
        \item $(-1, -2)$: $D_1 = -6, D_2 = 72$, ND $\Rightarrow$ \textbf{max}.
    \end{itemize}
    \item $f_x = 2 x y - y^2 + y = y(2 x - y + 1)$; $f_y = x^2 - 2 x y + x = x(x - 2 y + 1)$. Each product is zero when one factor is, giving four cases. $y = 0$, $x = 0$: $(0, 0)$. $y = 0$, $x - 2y + 1 = 0$: $x = -1$, so $(-1, 0)$. $x = 0$, $2x - y + 1 = 0$: $y = 1$, so $(0, 1)$. Both brackets zero: $y = 2x + 1$ and $x = 2y - 1 = 4x + 1$, so $x = -\tfrac13$, $y = \tfrac13$. Hessian: $H = \begin{pmatrix} 2 y & 2 x - 2 y + 1 \\ 2 x - 2 y + 1 & -2 x \end{pmatrix}$. At $(0,0)$: $H = \begin{pmatrix} 0 & 1 \\ 1 & 0 \end{pmatrix}$, $D_2 = -1 < 0$, \textbf{saddle}. At $(-1,0)$: $H = \begin{pmatrix} 0 & -1 \\ -1 & 2 \end{pmatrix}$, $D_2 = -1 < 0$, \textbf{saddle}. At $(0,1)$: $H = \begin{pmatrix} 2 & -1 \\ -1 & 0 \end{pmatrix}$, $D_2 = -1 < 0$, \textbf{saddle}. At $(-\tfrac13, \tfrac13)$: $2y = \tfrac23$, $-2x = \tfrac23$, $2x - 2y + 1 = -\tfrac13$; $D_1 = \tfrac23 > 0$, $D_2 = \tfrac49 - \tfrac19 = \tfrac13 > 0$, PD $\Rightarrow$ local \textbf{min}, $f(-\tfrac13, \tfrac13) = -\tfrac1{27}$. No critical point has $x, y > 0$: there both brackets would have to vanish, and subtracting them gives $x + y = 0$, impossible for positive $x, y$.
