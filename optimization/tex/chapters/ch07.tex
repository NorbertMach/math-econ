\chapter{The Lagrange Method: Setup and First-Order Conditions}

Real economic choices come with constraints: budgets, technology, time. The
technique of \emph{Lagrange multipliers} converts a constrained optimisation
problem into an unconstrained one in more variables. The first half of
mastering it is bookkeeping: pick the right sign convention, identify the
constraint constant, and write the FOCs without making algebra mistakes.
Once the Lagrangian and its FOCs are written down, the standard moves are:
divide two FOCs to eliminate $\lambda$, obtain an MRS condition between $x$
and $y$, substitute into the constraint, and back-solve for $\lambda^*$.
This chapter drills both halves (the setup and the solving pipeline)
with the canonical examples: Cobb--Douglas demand, linear constraints, and a
few non-textbook variants.

\section{Core Rules}

\begin{rulebox}{The standard problem}
Consider
\[
\max_{(x, y)} \; f(x, y) \quad \text{subject to} \quad g(x, y) = c,
\]
where $f$ is the objective (utility, output, profit) and $g(x, y) = c$ is a
single equality constraint (budget, resource).
\end{rulebox}

\begin{rulebox}{The Lagrangian and its sign convention}
Define the auxiliary function
\[
\mathcal{L}(x, y, \lambda) \;=\; f(x, y) + \lambda \bigl( c - g(x, y) \bigr).
\]
\textbf{Sign convention.} Writing it as $c - g$ (constraint constant minus
left-hand side) makes $\lambda \geq 0$ for natural maximisation problems,
relaxing the constraint by raising $c$ increases the optimum. Equivalent
forms appear in textbooks, e.g.\ $\mathcal{L} = f - \lambda(g - c)$ (the
same thing) or $\mathcal{L} = f + \lambda g$ when $c = 0$. The arithmetic is
identical; pick a convention and stick with it.
\end{rulebox}

\begin{rulebox}{First-order conditions}
Treat $\mathcal{L}$ as a function of three variables and set its partials to
zero:
\[
\frac{\partial \mathcal{L}}{\partial x} \;=\; f_x - \lambda g_x \;=\; 0, \qquad
\frac{\partial \mathcal{L}}{\partial y} \;=\; f_y - \lambda g_y \;=\; 0, \qquad
\frac{\partial \mathcal{L}}{\partial \lambda} \;=\; c - g(x, y) \;=\; 0.
\]
The first two equations say
\[
\grad f \;=\; \lambda \, \grad g \qquad \text{at the optimum.}
\]
Geometrically, the level sets of $f$ and $g$ are tangent there. The third
equation recovers the original constraint.
\end{rulebox}

\begin{rulebox}{Tangency in ratio form}
Dividing the first two FOCs (when $g_y \neq 0$ and $f_y \neq 0$) eliminates
$\lambda$:
\[
\frac{f_x}{f_y} \;=\; \frac{g_x}{g_y}.
\]
In consumer theory with $f = U(x, y)$ (utility) and $g = p_x x + p_y y$
(budget),
\[
\underbrace{\frac{U_x}{U_y}}_{\text{MRS}} \;=\; \frac{p_x}{p_y},
\]
the textbook equality of marginal rate of substitution and price ratio.

\textbf{Marginal rate of transformation.} For a general, possibly
nonlinear, constraint $g(x,y)=c$ (a production possibility frontier, an
opportunity curve) the slope of the constraint curve is
$\dd{y}{x}=-g_x/g_y$, and its absolute value
\[
  \text{MRT}=\frac{g_x}{g_y}
  \qquad\text{\de{Grenzrate der Transformation}}
\]
is the \emph{marginal rate of transformation}: how many units of $y$ must
be given up for one more unit of $x$ along the constraint. The tangency
condition is then the \textbf{economic optimality principle}
\[
  \text{MRS}=\frac{f_x}{f_y}=\frac{g_x}{g_y}=\text{MRT}:
\]
at the optimum the rate at which the decision maker is \emph{willing} to
trade $y$ for $x$ equals the rate at which technology or the market
\emph{allows} the trade. If MRS $>$ MRT (with $f_y>0$), moving along
the constraint towards more $x$ raises $f$, and conversely. With a linear budget the MRT
is the price ratio $p_x/p_y$.
\end{rulebox}

\begin{rulebox}{Identifying the constraint constant}
A budget $p_x x + p_y y = M$ has constraint constant $c = M$ and
$g(x,y) = p_x x + p_y y$. A production target $F(K, L) = Q_0$ has $c = Q_0$
and $g = F$. Always rewrite the constraint in the form $g(x,y) = c$ before
writing $\mathcal{L}$.
\end{rulebox}

\begin{rulebox}{When the method applies (constraint qualification)}
The FOC system presumes that $f$ and $g$ are differentiable and that
$\grad g \neq 0$ at the optimum, so that the tangency condition
$\grad f = \lambda \grad g$ is meaningful. When the objective is not
differentiable (e.g.\ the Leontief utility $\min\{x, y\}$, which has a
kink along the ray $x = y$) the standard Lagrangian does not apply and
the optimum must be found directly (see Practice Problem 7).
\end{rulebox}

\begin{rulebox}{Solving strategy: the four-step pipeline}
\begin{enumerate}[leftmargin=1.8em,itemsep=0.4ex]
  \item \textbf{Eliminate $\lambda$.} From $f_x = \lambda g_x$ and
        $f_y = \lambda g_y$ obtain
        \[
        \frac{f_x(x, y)}{f_y(x, y)} \;=\; \frac{g_x(x, y)}{g_y(x, y)}.
        \]
        For a linear budget $g = p_x x + p_y y$ this is
        $\dfrac{f_x}{f_y} = \dfrac{p_x}{p_y}$.
  \item \textbf{Solve the tangency for $y$ in terms of $x$ (or vice
        versa).} This produces the \emph{expansion path}, a relation
        between $x$ and $y$ that holds for any constraint level.
  \item \textbf{Plug into the constraint $g(x, y) = c$.} Two equations in
        two unknowns collapse to one in one unknown.
  \item \textbf{Back out $\lambda^*$.} Once $x^*, y^*$ are known,
        $\lambda^* = f_x(x^*, y^*)/g_x(x^*, y^*)$.
\end{enumerate}
\end{rulebox}

\begin{rulebox}{Worked template: Cobb--Douglas demand}
For $U = x^\alpha y^{1 - \alpha}$ and budget $p_x x + p_y y = M$:
\begin{enumerate}[leftmargin=1.8em,itemsep=0.4ex]
    \item FOCs: $\alpha x^{\alpha - 1} y^{1 - \alpha} = \lambda p_x$,
          $(1 - \alpha) x^\alpha y^{-\alpha} = \lambda p_y$.
    \item Divide: $\dfrac{\alpha}{1 - \alpha} \cdot \dfrac{y}{x} =
          \dfrac{p_x}{p_y}$, hence
          $y = \dfrac{(1 - \alpha) p_x}{\alpha p_y} x$.
    \item Budget: $p_x x + p_y \cdot \dfrac{(1 - \alpha) p_x}{\alpha p_y} x
          = M \Rightarrow p_x x \bigl(1 + \tfrac{1 - \alpha}{\alpha}\bigr)
          = M \Rightarrow p_x x / \alpha = M$.
    \item Hence $\boxed{x^* = \dfrac{\alpha M}{p_x}, \quad
          y^* = \dfrac{(1 - \alpha) M}{p_y}}$. A Cobb--Douglas consumer
          spends a constant share $\alpha$ on $x$ and $1 - \alpha$ on $y$.
\end{enumerate}
\end{rulebox}

\begin{rulebox}{The substitution method (and when to avoid it}
If the single equality constraint $g(x, y) = c$ can be solved explicitly for
one variable, say $y = h(x)$, substitute it into the objective and maximise
$f(x, h(x))$) an \emph{unconstrained} problem in one variable fewer. This
always works when the solve-out is global and smooth. But beware:
\begin{itemize}[leftmargin=1.8em,itemsep=0.4ex]
  \item the constraint may not be solvable in closed form;
  \item solving for one variable can lose branches (e.g.\
        $y = \pm\sqrt{c - x^2}$) or hide domain restrictions;
  \item corner information and the shadow price $\lambda$ are lost, the
        Lagrangian keeps both;
  \item with several constraints, substitution quickly becomes impractical.
\end{itemize}
Rule of thumb: substitution for quick checks in two variables; Lagrange as
the systematic method.
\end{rulebox}

\begin{rulebox}{Graphical solution}
For two variables the Lagrange problem can be solved, or at least
understood, in a picture:
\begin{enumerate}[leftmargin=1.8em,itemsep=0.3ex]
  \item Draw the constraint curve $g(x,y)=c$ and several level curves
        $f(x,y)=k$.
  \item Move the level $k$ in the improving direction (the direction of
        $\grad f$ for a maximum). The optimum is the last level curve that
        still meets the constraint; there the two curves \emph{touch}.
  \item At the touching point $P$ the gradients are parallel,
        $\grad f(P)=\lambda\grad g(P)$, and the ratio of their lengths is
        $|\lambda|=\|\grad f(P)\|/\|\grad g(P)\|$. Same direction:
        $\lambda>0$; opposite directions: $\lambda<0$.
  \item The level of the curve through a critical point \emph{is} the
        critical value $f(P)$. Comparing the levels through all critical
        points tells maximum from minimum; on a closed bounded constraint
        curve (circle, ellipse) both exist by Weierstrass.
\end{enumerate}
A level curve that crosses the constraint instead of touching it is never
optimal: moving along the constraint on one side raises $f$, on the other
side lowers it.
\end{rulebox}

\begin{rulebox}{Writing the budget as an equality; existence; model critique}
Economic constraints usually read ``spend \emph{at most} $M$'', i.e.\
$p_xx+p_yy\le M$. Three questions belong to every such problem.
\begin{itemize}[leftmargin=1.6em,itemsep=0.3ex]
  \item \textbf{Why may the inequality be written as an equality?} If the
        objective is strictly increasing in at least one good with a
        positive price (no satiation), any bundle that leaves money unspent
        can be improved by buying more of that good. So the budget binds at
        the optimum and the Lagrange method applies to $p_xx+p_yy=M$. With
        a satiation point inside the budget set this fails, and the
        equality form produces a wrong answer with $\lambda^*<0$.
  \item \textbf{Does an optimum exist?} With $p_x,p_y,M>0$ the budget set
        $\{p_xx+p_yy\le M,\ x,y\ge0\}$ is closed and bounded (a triangle);
        a continuous objective attains its maximum there (Weierstrass,
        \chref{optimization}{1}). A zero price or a missing sign restriction makes the
        set unbounded, and existence must be argued separately.
  \item \textbf{What is unrealistic?} Typical idealisations: goods and
        inputs perfectly divisible; prices independent of the quantity
        bought; no fixed or set-up costs; every unit of budget usable;
        smooth substitution between inputs; no satiation. Naming them is
        part of a complete answer.
\end{itemize}
Finally, check the \textbf{domain}: a Lagrange or substitution candidate
can fall outside the meaningful region (a negative quantity or length).
Then the optimum lies on the boundary of the domain, and the substitution
route with the domain stated explicitly, plus a sketch of the
one-variable function, finds it.
\end{rulebox}

\begin{rulebox}{Mean--variance portfolio choice}
Invest the share $x$ of wealth in asset $A$ and $1-x$ in asset $B$, with
expected returns $\mu_A,\mu_B$, variances $\sigma_A^{2},\sigma_B^{2}$ and
covariance $\sigma_{AB}$. The portfolio has
\[
  \mu=x\mu_A+(1-x)\mu_B,\qquad
  \sigma^{2}=x^{2}\sigma_A^{2}+(1-x)^{2}\sigma_B^{2}+2x(1-x)\sigma_{AB}.
\]
\begin{enumerate}[leftmargin=1.8em,itemsep=0.3ex]
  \item \textbf{Opportunity curve.} Eliminate
        $x=\dfrac{\mu-\mu_B}{\mu_A-\mu_B}$: the attainable pairs satisfy
        $\sigma^{2}=a\mu^{2}+b\mu+c$ with $a>0$, a parabola in the
        $(\mu,\sigma^{2})$ plane. Its vertex is the minimum-variance
        portfolio; only the branch above it is efficient.
  \item \textbf{Preferences.} The investor maximises $\Phi(\mu,\sigma)$,
        increasing in $\mu$ and decreasing in risk, e.g.\
        $\mu-k\sigma^{2}$, $\mu-k\sigma$ or $\mu-k(\mu^{2}+\sigma^{2})$,
        subject to the opportunity curve. Use the variables $(\mu,s)$ with
        $s=\sigma^{2}$ when $\Phi$ contains $\sigma^{2}$; when it contains
        $\sigma$, use $(\mu,\sigma)$ and the constraint
        $\sigma^{2}=a\mu^{2}+b\mu+c$, or differentiate
        $\sigma=\sqrt{a\mu^{2}+b\mu+c}$ by the chain rule.
  \item \textbf{Back out the portfolio.} From $\mu^*$ get $x^*$, then the
        money amounts $x^*W$ and $(1-x^*)W$ for wealth $W$.
  \item \textbf{MRS $=$ MRT in the $(\mu,\sigma)$ plane.} The slope
        $\mathrm{d}\mu/\mathrm{d}\sigma$ of the indifference curve through
        the optimum equals that of the opportunity curve.
  \item \textbf{Global.} Substituting the constraint into $\Phi$ usually
        gives a concave function of $\mu$ (a concave $\Phi$ minus a convex
        risk term), so the critical point is the global maximum.
\end{enumerate}
\end{rulebox}

\section{Worked Examples}

\begin{worked}{Cost minimisation: minimise $wL+rK$ subject to $K^{1/2}L^{1/2}=\bar Q$}
\wstep{Step 1. Set up the Lagrangian --- and mind the sign convention for a minimum.}
Write the constraint as ``constant minus function equals zero'' and attach
the multiplier:
\[
  \mathcal L(L,K,\lambda)=wL+rK+\lambda\bigl(\bar Q-K^{1/2}L^{1/2}\bigr).
\]
\wtrap{Any consistent convention works --- $+\lambda(\bar Q-g)$ or
$-\lambda(g-\bar Q)$ --- provided you keep it for all conditions. What
changes with the convention is only the \emph{sign} of $\lambda$, and you
must know which sign you have chosen before interpreting it. Here
$\lambda>0$ will mean ``cost rises when required output rises'', which is
what we want.}

\wstep{Step 2. First-order conditions.}
\[
  \pd{\mathcal L}{L}=w-\lambda\cdot\tfrac12K^{1/2}L^{-1/2}=0,
  \qquad
  \pd{\mathcal L}{K}=r-\lambda\cdot\tfrac12K^{-1/2}L^{1/2}=0,
\]
\[
  \pd{\mathcal L}{\lambda}=\bar Q-K^{1/2}L^{1/2}=0 .
\]
Three equations, three unknowns. The third is just the constraint restated,
which is the point of differentiating with respect to $\lambda$.

\wstep{Step 3. Divide the first two to eliminate $\lambda$.}
\[
  \frac{w}{r}
  =\frac{\tfrac12K^{1/2}L^{-1/2}}{\tfrac12K^{-1/2}L^{1/2}}
  =K^{1/2-(-1/2)}L^{-1/2-1/2}
  =\frac{K}{L} .
\]
\[
  \boxed{\frac{K}{L}=\frac{w}{r}}
\]
\wtrap{The exponent arithmetic is where errors occur: $\tfrac12-(-\tfrac12)=1$
and $-\tfrac12-\tfrac12=-1$. Write both subtractions with brackets. The
reward is that all the awkward powers collapse into the single ratio $K/L$
--- which is what makes the rest of the problem easy.}

\wstep{Step 4. Read the condition before using it.}
It says $MRTS=$ input price ratio: the isoquant is tangent to the isocost
line. Every constrained problem in this volume produces a tangency
condition at this stage, and recognising it confirms the algebra is on
track. Note it is the \emph{opposite} orientation to the consumer problem
--- there the objective is curved and the constraint straight, here the
constraint is curved and the objective straight --- but the tangency is the
same.

\wstep{Step 5. Impose the constraint.}
From $K=(w/r)L$,
\[
  \bar Q=K^{1/2}L^{1/2}=\Bigl(\tfrac wr L\Bigr)^{1/2}L^{1/2}
        =L\sqrt{\tfrac wr}
  \quad\Longrightarrow\quad
  L^{*}=\bar Q\sqrt{\frac rw},
  \qquad
  K^{*}=\bar Q\sqrt{\frac wr} .
\]

\wstep{Step 6. The cost function.}
\[
  C(\bar Q,w,r)=wL^{*}+rK^{*}
  =w\bar Q\sqrt{\tfrac rw}+r\bar Q\sqrt{\tfrac wr}
  =\bar Q\sqrt{wr}+\bar Q\sqrt{wr}
  =\boxed{2\bar Q\sqrt{wr}}
\]
Each input absorbs exactly half the cost --- the shares are the exponents
$\tfrac12,\tfrac12$, as always for Cobb--Douglas.

\wstep{Step 7. Recover $\lambda$ and check what it means.}
From the first condition,
\[
  \lambda=\frac{w}{\tfrac12K^{1/2}L^{-1/2}}
  =\frac{2w}{\sqrt{K/L}}
  =\frac{2w}{\sqrt{w/r}}
  =2\sqrt{wr} .
\]
And differentiating the cost function directly,
\[
  \pd{C}{\bar Q}=2\sqrt{wr}=\lambda . \checkmark
\]
So $\lambda$ is the \textbf{marginal cost} of output --- the shadow price of
the constraint. It is constant here because the technology has constant
returns to scale.

\wstep{Step 8. Verify the answer is a minimum, not a maximum.}
The first-order conditions are necessary for either. Here the argument is
structural rather than computational: the objective $wL+rK$ is linear (hence
convex), and the feasible set $\{K^{1/2}L^{1/2}\ge\bar Q\}$ is convex,
because $K^{1/2}L^{1/2}$ is concave and this is a superlevel set. A convex
objective over a convex set has its stationary point as a global minimum.
\wtrap{Lagrange finds stationary points of $\mathcal L$, which include
maxima, minima and saddles alike. Some argument (a bordered Hessian,
\chref{optimization}{9}, or convexity as here) is always required. And check the
constraint qualification: $\grad g=(\tfrac12K^{-1/2}L^{1/2},
\tfrac12K^{1/2}L^{-1/2})\neq\mathbf0$ for $K,L>0$, so the method
applies.}

\wstep{Step 9. The four-step pipeline, for reuse.}
\begin{center}\footnotesize
\begin{tabular}{ll}
  \hline
  1 & write $\mathcal L$, fixing a sign convention\\
  2 & set all partials to zero, including the one in $\lambda$\\
  3 & divide two conditions to eliminate $\lambda$ $\to$ tangency\\
  4 & substitute into the constraint, solve, then recover $\lambda$\\
  \hline
\end{tabular}
\end{center}
\end{worked}

\begin{worked}{Mean--variance portfolio: minimise the variance of three uncorrelated assets for a target return of $4\%$, then let an investor choose the target}
\wstep{Step 1. The data and the problem.}
Three assets $A,B,C$ have expected returns $\mu=(2,4,8)$ (in \%),
variances $\sigma^{2}=(2,8,16)$ (in $\%^{2}$) and zero covariances. With
portfolio weights $w_A,w_B,w_C$ the portfolio variance is
$V=2w_A^{2}+8w_B^{2}+16w_C^{2}$. The investor wants the safest portfolio
with expected return $4\%$:
\[
  \min V \quad\text{s.t.}\quad 2w_A+4w_B+8w_C=4
  \quad\text{(target return)},\qquad
  w_A+w_B+w_C=1 \quad\text{(budget)} .
\]
With two assets the two constraints would already fix the weights (the
opportunity curve of the portfolio box). With three, a whole line of
portfolios meets both, and the variance picks one of them.

\wstep{Step 2. One multiplier per constraint.}
Keep the convention ``constant minus left-hand side'':
\[
  \mathcal L=2w_A^{2}+8w_B^{2}+16w_C^{2}
  +\lambda\bigl(4-2w_A-4w_B-8w_C\bigr)
  +\gamma\bigl(1-w_A-w_B-w_C\bigr).
\]
A second constraint simply brings a second multiplier and a second
$\partial\mathcal L/\partial(\cdot)=0$ equation; \chref{optimization}{9}
develops this in general.

\wstep{Step 3. First-order conditions: each weight in terms of the multipliers.}
\[
  4w_A-2\lambda-\gamma=0,\qquad
  16w_B-4\lambda-\gamma=0,\qquad
  32w_C-8\lambda-\gamma=0,
\]
so
\[
  w_A=\frac{2\lambda+\gamma}{4},\qquad
  w_B=\frac{4\lambda+\gamma}{16},\qquad
  w_C=\frac{8\lambda+\gamma}{32}.
\]
In general $w_i=(\lambda\mu_i+\gamma)/(2\sigma_i^{2})$: a high return
attracts weight, a high variance repels it.

\wstep{Step 4. Substitute into the two constraints.}
Return (multiply by $4$): $(4\lambda+2\gamma)+(4\lambda+\gamma)
+(8\lambda+\gamma)=16$, i.e.\ $4\lambda+\gamma=4$. Budget (multiply by
$32$): $(16\lambda+8\gamma)+(8\lambda+2\gamma)+(8\lambda+\gamma)=32$, i.e.\
$32\lambda+11\gamma=32$. With $\gamma=4-4\lambda$: $32\lambda+44-44\lambda=32$,
so
\[
  \lambda^{*}=1,\qquad \gamma^{*}=0,\qquad
  (w_A^{*},w_B^{*},w_C^{*})=\Bigl(\tfrac12,\tfrac14,\tfrac14\Bigr),
\]
\[
  V^{*}=2\cdot\tfrac14+8\cdot\tfrac1{16}+16\cdot\tfrac1{16}=2,
  \qquad \sigma^{*}=\sqrt2\approx1.41\% .
\]
Check: $\tfrac12+\tfrac14+\tfrac14=1$ and $1+1+2=4$. \checkmark\ Asset $B$
alone also returns $4\%$, but with variance $8$: diversification cuts the
variance to a quarter.

\wstep{Step 5. The whole frontier, and what $\lambda$ measures.}
Repeat Step~4 with a general target $m$ in place of $4$:
$4\lambda+\gamma=m$ and $32\lambda+11\gamma=32$ give
\[
  \lambda=\frac{11m-32}{12},\qquad \gamma=\frac{32-8m}{3},\qquad
  V^{*}(m)=\frac{11}{24}m^{2}-\frac83m+\frac{16}{3}.
\]
This is the opportunity curve $\sigma^{2}=am^{2}+bm+c$ of the portfolio box,
now for three assets. Its vertex, the minimum-variance portfolio, is at
$m=\tfrac{32}{11}\approx2.91\%$ with $V=\tfrac{16}{11}\approx1.45$. And
$\mathrm{d}V^{*}/\mathrm{d}m=\tfrac{11}{12}m-\tfrac83$ equals $1$ at $m=4$:
the multiplier is the marginal variance of the target, as
$\partial\mathcal L/\partial m=\lambda$ predicts. Raising the target from
$4\%$ to $4.1\%$ costs about $0.1$ units of variance.
\wtrap{The sign of $\lambda$ carries information. For a target below
$\tfrac{32}{11}\%$ the formula gives $\lambda<0$: tightening the return
requirement would \emph{lower} the variance. Such a portfolio lies on the
inefficient lower branch, dominated by the minimum-variance portfolio,
which has a higher return and a lower variance. Before reporting a
frontier portfolio, check $\lambda\ge0$.}

\wstep{Step 6. Global, not just stationary.}
$V$ has the diagonal Hessian $\operatorname{diag}(4,16,32)$, positive
definite, so $V$ is strictly convex; both constraints are linear, so the
feasible set is a line, a convex set. A strictly convex objective on a
convex set has its stationary point as the unique global minimum: no
bordered Hessian is needed.

\wstep{Step 7. Which target? The investor's choice (box item 2).}
An investor with $\Phi(m,s)=m-s$ (here $k=1$, $s=\sigma^{2}$) maximises
$\Phi$ along the frontier:
\[
  \mathcal L=m-s+\nu\Bigl(s-\tfrac{11}{24}m^{2}+\tfrac83m-\tfrac{16}{3}\Bigr),
  \qquad
  \pd{\mathcal L}{s}=-1+\nu=0,\qquad
  \pd{\mathcal L}{m}=1-\nu\Bigl(\tfrac{11}{12}m-\tfrac83\Bigr)=0 .
\]
So $\nu=1$ and $\tfrac{11}{12}m=\tfrac{11}{3}$: $m^{*}=4$, the target of
Step~1, with $s^{*}=2$ and $\Phi^{*}=2$. This is MRS $=$ MRT: the frontier
slope $\mathrm{d}s/\mathrm{d}m=1$ equals the investor's exchange rate
$1/k=1$ between return and variance. The substituted objective
$m-V^{*}(m)$ is concave, so this is the global maximum.

\wstep{Step 8. Back out the money amounts (box item 3).}
With wealth $W=10\,000$ euros: $5\,000$ in $A$, $2\,500$ in $B$,
$2\,500$ in $C$, for an expected return of $400$ euros and a standard
deviation of about $141$ euros.
\end{worked}

For Problems 1--15 we record $\mathcal{L}$ and the three FOCs.

\begin{solutions}
  \item $\mathcal{L} = x y + \lambda(12 - 2 x - 3 y)$.
  \[
  \mathcal{L}_x = y - 2 \lambda = 0, \quad \mathcal{L}_y = x - 3 \lambda = 0, \quad \mathcal{L}_\lambda = 12 - 2 x - 3 y = 0.
  \]
  \item $\mathcal{L} = x^{1/2} y^{1/2} + \lambda(M - p_x x - p_y y)$.
  \[
  \mathcal{L}_x = \tfrac{1}{2} x^{-1/2} y^{1/2} - \lambda p_x = 0, \quad \mathcal{L}_y = \tfrac{1}{2} x^{1/2} y^{-1/2} - \lambda p_y = 0, \quad \mathcal{L}_\lambda = M - p_x x - p_y y = 0.
  \]
  \item $\mathcal{L} = x^\alpha y^{1 - \alpha} + \lambda(M - p_x x - p_y y)$.
  \[
  \mathcal{L}_x = \alpha x^{\alpha - 1} y^{1 - \alpha} - \lambda p_x = 0, \quad \mathcal{L}_y = (1 - \alpha) x^\alpha y^{-\alpha} - \lambda p_y = 0, \quad \mathcal{L}_\lambda = M - p_x x - p_y y = 0.
  \]
  \item $\mathcal{L} = \ln(x) + \ln(y) + \lambda(10 - x - y)$.
  \[
  \mathcal{L}_x = \tfrac{1}{x} - \lambda = 0, \quad \mathcal{L}_y = \tfrac{1}{y} - \lambda = 0, \quad \mathcal{L}_\lambda = 10 - x - y = 0.
  \]
  \item $\mathcal{L} = \ln(x) + 2 \ln(y) + \lambda(24 - x - 4 y)$.
  \[
  \mathcal{L}_x = \tfrac{1}{x} - \lambda = 0, \quad \mathcal{L}_y = \tfrac{2}{y} - 4 \lambda = 0, \quad \mathcal{L}_\lambda = 24 - x - 4 y = 0.
  \]
  \item $\mathcal{L} = (x^\rho + y^\rho)^{1/\rho} + \lambda(M - p_x x - p_y y)$. With $U = (x^\rho + y^\rho)^{1/\rho}$, $U_x = U^{1 - \rho} x^{\rho - 1}$ (chain rule: $U_x = \tfrac{1}{\rho}(x^\rho + y^\rho)^{1/\rho - 1} \cdot \rho x^{\rho - 1} = U^{1 - \rho} x^{\rho - 1}$).
  \[
  \mathcal{L}_x = U^{1 - \rho} x^{\rho - 1} - \lambda p_x = 0, \quad \mathcal{L}_y = U^{1 - \rho} y^{\rho - 1} - \lambda p_y = 0, \quad \mathcal{L}_\lambda = M - p_x x - p_y y = 0.
  \]
  \item The Leontief utility $\min\{x, y\}$ is not differentiable along the ray $x = y$, so $\mathcal{L}_x$ and $\mathcal{L}_y$ do not exist there. The standard FOC method fails. The shortcut: optimal consumption requires $x = y$ (no waste), so substitute into the budget: $p_x x + p_y x = M \Rightarrow x^* = y^* = M/(p_x + p_y)$. (More generally, for $\min\{x/a, y/b\}$ set $x/a = y/b$.)
  \item $\mathcal{L} = p_K K + p_L L + \lambda(Q_0 - K^{1/2} L^{1/2})$. \emph{(Minimisation; same Lagrangian shape, the multiplier still has the right sign because we wrote constraint as $Q_0 - F$.)}
  \[
  \mathcal{L}_K = p_K - \tfrac{\lambda}{2} K^{-1/2} L^{1/2} = 0, \quad \mathcal{L}_L = p_L - \tfrac{\lambda}{2} K^{1/2} L^{-1/2} = 0, \quad \mathcal{L}_\lambda = Q_0 - K^{1/2} L^{1/2} = 0.
  \]
  \item $\mathcal{L} = x^2 + y^2 + \lambda(4 - x - y)$.
  \[
  \mathcal{L}_x = 2 x - \lambda = 0, \quad \mathcal{L}_y = 2 y - \lambda = 0, \quad \mathcal{L}_\lambda = 4 - x - y = 0.
  \]
  \item $\mathcal{L} = K^{1/3} L^{2/3} + \lambda(TC - r K - w L)$.
  \[
  \mathcal{L}_K = \tfrac{1}{3} K^{-2/3} L^{2/3} - \lambda r = 0, \quad \mathcal{L}_L = \tfrac{2}{3} K^{1/3} L^{-1/3} - \lambda w = 0, \quad \mathcal{L}_\lambda = TC - r K - w L = 0.
  \]
  \item $\mathcal{L} = 2 x + 3 y + \lambda(13 - x^2 - y^2)$.
  \[
  \mathcal{L}_x = 2 - 2 \lambda x = 0, \quad \mathcal{L}_y = 3 - 2 \lambda y = 0, \quad \mathcal{L}_\lambda = 13 - x^2 - y^2 = 0.
  \]
  \item $\mathcal{L} = x y^2 + \lambda(9 - x - y)$.
  \[
  \mathcal{L}_x = y^2 - \lambda = 0, \quad \mathcal{L}_y = 2 x y - \lambda = 0, \quad \mathcal{L}_\lambda = 9 - x - y = 0.
  \]
  \item $\mathcal{L} = \alpha \ln(x) + \beta \ln(y) + \lambda(M - p_x x - p_y y)$.
  \[
  \mathcal{L}_x = \tfrac{\alpha}{x} - \lambda p_x = 0, \quad \mathcal{L}_y = \tfrac{\beta}{y} - \lambda p_y = 0, \quad \mathcal{L}_\lambda = M - p_x x - p_y y = 0.
  \]
  \item $\mathcal{L} = x^2 + 4 y^2 + \lambda(6 - x - 2 y)$.
  \[
  \mathcal{L}_x = 2 x - \lambda = 0, \quad \mathcal{L}_y = 8 y - 2 \lambda = 0, \quad \mathcal{L}_\lambda = 6 - x - 2 y = 0.
  \]
  \item $\mathcal{L} = x y + \lambda(8 - x^2 - y^2)$.
  \[
  \mathcal{L}_x = y - 2 \lambda x = 0, \quad \mathcal{L}_y = x - 2 \lambda y = 0, \quad \mathcal{L}_\lambda = 8 - x^2 - y^2 = 0.
  \]
  \item FOCs: $y = 2 \lambda$, $x = 3 \lambda$. Divide: $y/x = 2/3 \Rightarrow x = 3 y / 2$. Sub: $2 (3 y/2) + 3 y = 6 y = 12 \Rightarrow y^* = 2$, $x^* = 3$. $\lambda^* = y/2 = 1$. $U^* = 6$.
  \item FOCs: $\tfrac{1}{2} x^{-1/2} y^{1/2} = \lambda p_x$, $\tfrac{1}{2} x^{1/2} y^{-1/2} = \lambda p_y$. Divide: $y/x = p_x/p_y$. So $y = (p_x/p_y) x$. Budget: $p_x x + p_y (p_x/p_y) x = 2 p_x x = M \Rightarrow x^* = \dfrac{M}{2 p_x}$, $y^* = \dfrac{M}{2 p_y}$. Equal-share spending (since this is Cobb--Douglas with $\alpha = 1/2$). $\lambda^* = \dfrac{1}{2 p_x} \cdot \dfrac{y^{*1/2}}{x^{*1/2}} = \dfrac{1}{2 p_x} \sqrt{p_x/p_y} = \dfrac{1}{2\sqrt{p_x p_y}}$.
  \item FOCs: $1/x = \lambda$, $1/y = \lambda \Rightarrow x = y$. Constraint: $2 x = 10 \Rightarrow x^* = y^* = 5$. $\lambda^* = 1/5$.
  \item FOCs: $1/x = \lambda$ and $2/y = 4 \lambda \Rightarrow y = 2/(4 \lambda) = 1/(2 \lambda)$. From $1/x = \lambda$ we have $\lambda = 1/x$. Then $2/y = 4/x \Rightarrow y = x/2$. Sub into $x + 4 y = 24$: $x + 2 x = 3 x = 24 \Rightarrow x^* = 8$, $y^* = 4$. $\lambda^* = 1/8$.
  \item FOCs: $y^2 = \lambda$, $2 x y = \lambda$. So $y^2 = 2 x y \Rightarrow y = 2 x$ (taking $y > 0$). Sub into $x + y = 9$: $3 x = 9 \Rightarrow x^* = 3$, $y^* = 6$. $\lambda^* = 36$. $U^* = 3 \cdot 36 = 108$.
  \item FOCs: $\alpha/x = \lambda p_x$, $\beta/y = \lambda p_y$. So $p_x x = \alpha/\lambda$ and $p_y y = \beta/\lambda$. Sum: $p_x x + p_y y = (\alpha + \beta)/\lambda = M \Rightarrow \lambda^* = (\alpha + \beta)/M$. Hence
  \[ x^* = \frac{\alpha}{\alpha + \beta} \cdot \frac{M}{p_x}, \quad y^* = \frac{\beta}{\alpha + \beta} \cdot \frac{M}{p_y}. \]
  Constant expenditure shares $\alpha/(\alpha + \beta)$ and $\beta/(\alpha + \beta)$.
  \item FOCs: $2 = 2 \lambda x$, $3 = 2 \lambda y$, so $x = 1/\lambda$, $y = 3/(2 \lambda)$. Constraint: $1/\lambda^2 + 9/(4 \lambda^2) = 13 \Rightarrow (4 + 9)/(4 \lambda^2) = 13 \Rightarrow \lambda^2 = 1/4 \Rightarrow \lambda^* = 1/2$ (taking positive root for max). Then $x^* = 2$, $y^* = 3$. $f^* = 4 + 9 = 13$.
  \item FOCs: $y = 2 \lambda x$, $x = 2 \lambda y$. Multiply: $x y = 4 \lambda^2 x y$, so (with $x, y > 0$) $\lambda^2 = 1/4 \Rightarrow \lambda^* = 1/2$. Then $y = x$. Constraint: $2 x^2 = 8 \Rightarrow x^* = 2$, $y^* = 2$. $f^* = 4$.
  \item FOCs: $2 x = \lambda$, $2 y = \lambda \Rightarrow x = y$. Constraint: $2 x = 4 \Rightarrow x^* = y^* = 2$. $\lambda^* = 4$. $f^* = 8$.
  \item FOCs: $2 x = \lambda$, $8 y = 2 \lambda \Rightarrow \lambda = 4 y$. Equate: $2 x = 4 y \Rightarrow x = 2 y$. Constraint: $2 y + 2 y = 4 y = 6 \Rightarrow y^* = 3/2$, $x^* = 3$. $\lambda^* = 6$. $f^* = 9 + 9 = 18$.
  \item FOCs: $\tfrac{1}{2} K^{-1/2} L^{1/2} = \lambda r$, $\tfrac{1}{2} K^{1/2} L^{-1/2} = \lambda w$. Divide: $L/K = r/w \Rightarrow w L = r K$. With $r K + w L = TC$: $2 r K = TC \Rightarrow K^* = \dfrac{TC}{2 r}$, $L^* = \dfrac{TC}{2 w}$. $Q^* = (K^* L^*)^{1/2} = \dfrac{TC}{2 \sqrt{r w}}$.
  \item Now $C$ is the objective and $K^{1/2} L^{1/2} = Q_0$ is the constraint. $\mathcal{L} = p_K K + p_L L + \lambda(Q_0 - K^{1/2} L^{1/2})$. FOCs: $p_K = \tfrac{\lambda}{2} K^{-1/2} L^{1/2}$, $p_L = \tfrac{\lambda}{2} K^{1/2} L^{-1/2}$. Divide: $p_K/p_L = L/K \Rightarrow L = (p_K/p_L) K$. Constraint: $K^{1/2}(p_K/p_L)^{1/2} K^{1/2} = Q_0 \Rightarrow K \sqrt{p_K/p_L} = Q_0 \Rightarrow K^* = Q_0 \sqrt{p_L/p_K}$, $L^* = Q_0 \sqrt{p_K/p_L}$. Minimum cost: $C^* = p_K K^* + p_L L^* = 2 Q_0 \sqrt{p_K p_L}$.
  \item FOCs: $\tfrac{1}{3} K^{-2/3} L^{2/3} = \lambda r$, $\tfrac{2}{3} K^{1/3} L^{-1/3} = \lambda w$. Divide: $\dfrac{1/3 \cdot L/K}{2/3} = r/w \Rightarrow \dfrac{L}{2 K} = r/w \Rightarrow w L = 2 r K$. Constraint: $r K + 2 r K = 3 r K = TC \Rightarrow K^* = \dfrac{TC}{3 r}$, $L^* = \dfrac{2 TC}{3 w}$.
  \item FOCs: $4/x = 2 \lambda$, $3/y = \lambda$. So $\lambda = 2/x$ and $\lambda = 3/y \Rightarrow y = 3 x / 2$. Constraint: $2 x + 3 x/2 = 7 x/2 = 35 \Rightarrow x^* = 10$, $y^* = 15$. $\lambda^* = 1/5$.
  \item $\mathcal{L} = x y + 2 x + \lambda(8 - x - y)$. FOCs: $y + 2 = \lambda$, $x = \lambda$. So $x = y + 2$. Constraint: $(y + 2) + y = 2 y + 2 = 8 \Rightarrow y^* = 3$, $x^* = 5$. $\lambda^* = 5$. $U^* = 15 + 10 = 25$.
  \item $\mathcal{L} = x + 4 \sqrt{y} + \lambda(10 - x - y)$. FOCs: $1 = \lambda$, $2/\sqrt{y} = \lambda = 1 \Rightarrow \sqrt{y} = 2 \Rightarrow y^* = 4$. Constraint: $x^* = 6$. $U^* = 6 + 4 \cdot 2 = 14$. $\lambda^* = 1$.
  \item FOCs: $6 x = \lambda$, $4 y = \lambda \Rightarrow 6 x = 4 y \Rightarrow y = 3 x / 2$. Constraint: $x + 3 x/2 = 5 x/2 = 5 \Rightarrow x^* = 2$, $y^* = 3$. $\lambda^* = 12$. $f^* = 12 + 18 = 30$.
  \item FOCs: $\tfrac{1}{2 \sqrt{x}} = \lambda p_x$, $\tfrac{1}{2 \sqrt{y}} = \lambda p_y$. Divide: $\sqrt{y/x} = p_x/p_y \Rightarrow y = (p_x/p_y)^2 x$. Sub: $p_x x + p_y (p_x/p_y)^2 x = p_x x + (p_x^2/p_y) x = x \cdot p_x(1 + p_x/p_y) = x \cdot p_x (p_x + p_y)/p_y = M$. Hence
  \[ x^* = \frac{M p_y}{p_x(p_x + p_y)}, \qquad y^* = \left(\frac{p_x}{p_y}\right)^2 x^* = \frac{M p_x}{p_y(p_x + p_y)}. \]
\end{solutions}

          =\frac{35\mu^{2}-340\mu+1100}{3}.
        \]
        (c) $\tfrac{\mathrm{d}\sigma^{2}}{\mathrm{d}x}=840x-120=0$:
        $x=\tfrac17$, $\mu=4+\tfrac67=\tfrac{34}{7}\approx4.86$,
        $\sigma^{2}=\tfrac{420}{49}-\tfrac{120}{7}+100=\tfrac{640}{7}\approx91.4$,

        $\sigma_B/\sigma_A=0.5$.
  \item (a) With $s=\sigma^{2}$ and
        $\mathcal{L}=\mu-0.01s-\lambda\bigl(s-\tfrac{35\mu^{2}-340\mu+1100}{3}\bigr)$:

        $\mathcal{L}_\mu=1+\lambda\,\tfrac{70\mu-340}{3}=0$ gives
        $70\mu-340=300$, $\mu^*=\tfrac{64}{7}\approx9.14\%$, and
        $s^*=\tfrac{2140}{7}\approx305.7$, $\sigma^*\approx17.48\%$,
        $\Phi^*=\tfrac{213}{35}\approx6.09$.
        (b) $x^*=\tfrac{\mu^*-4}{6}=\tfrac67$: $60\,000$ in $A$, $10\,000$ in

        (c) Indifference curve $\mu-0.01\sigma^{2}=\text{const}$:
        $\tfrac{\mathrm{d}\mu}{\mathrm{d}\sigma}=0.02\sigma$. Opportunity
        curve: $2\sigma\,\mathrm{d}\sigma=\tfrac{70\mu-340}{3}\mathrm{d}\mu
        =100\,\mathrm{d}\mu$ at $\mu^*$, so
        $\tfrac{\mathrm{d}\mu}{\mathrm{d}\sigma}=\tfrac{2\sigma}{100}
        =0.02\sigma$. Equal ($\approx0.350$ at $\sigma^*$).
        (d) On the curve $\Phi(\mu)=\mu-\tfrac{35\mu^{2}-340\mu+1100}{300}$,

  \item With $\mathcal{L}=\mu-\tfrac54\sigma-\lambda\bigl(\sigma^{2}-\mu^{2}+8\mu-25\bigr)$:
        $\mathcal{L}_\mu=1+\lambda(2\mu-8)=0$ and
        $\mathcal{L}_\sigma=-\tfrac54-2\lambda\sigma=0$, so
        $\lambda=-\tfrac{5}{8\sigma}$ and then
        $1=\tfrac{5(2\mu-8)}{8\sigma}$, i.e.\ $\sigma=\tfrac54(\mu-4)$.
        The constraint: $\tfrac{25}{16}(\mu-4)^{2}=(\mu-4)^{2}+9$, so
        $(\mu-4)^{2}=16$; $\sigma>0$ needs $\mu>4$: $\mu^*=8$, $\sigma^*=5$,
        $\lambda^*=-\tfrac18$, $\Phi^*=8-\tfrac{25}{4}=\tfrac74$. Global: on
        the efficient branch $\sigma(\mu)=\sqrt{(\mu-4)^{2}+9}$ is the length
        of the vector $(\mu-4,3)$, a convex function of $\mu$, so
        $\Phi(\mu)=\mu-\tfrac54\sigma(\mu)$ is concave. With a coefficient
        $k\le1$, $\Phi'(\mu)=1-k\tfrac{\mu-4}{\sigma}>0$ for all $\mu$ (since
        $\mu-4<\sigma$): $\Phi$ keeps rising and no optimum exists, because
        far out the curve grows only like $\sigma\approx\mu-4$.
  \item On the curve,
        $\Phi(\mu)=\mu-\tfrac{\mu^{2}}{100}-\tfrac{35\mu^{2}-340\mu+1100}{300}$.

  \item $\sigma^{2}(x)=225x^{2}+64(1-x)^{2}-48x(1-x)=337x^{2}-176x+64$.

  \item (a) The tangency condition becomes

  \item (a) $\mathcal{L}=x+2\ln y+\lambda(M-x-y)$: $1=\lambda$,
