\chapter{Eigenvalues and Eigenvectors}

An eigenvector of $A \in \R^{n \times n}$ is a direction that the linear map
$x \mapsto A x$ leaves invariant (up to scaling). The corresponding scale
factor is the eigenvalue. Eigenvalues encode long-run behaviour of dynamical
systems, drive the spectral characterisation of definiteness used for
second-order conditions, and provide the means to compute powers $A^{n}$
efficiently via diagonalisation.

\section{Core Rules}

\begin{rulebox}{Definition}
A scalar $\lambda \in \R$ (or $\mathbb{C}$) is an \emph{eigenvalue} of
$A \in \R^{n \times n}$ if there exists a nonzero vector $v$ with
\[
A v \;=\; \lambda v.
\]
Such a $v$ is an \emph{eigenvector} for $\lambda$.
\end{rulebox}

\begin{rulebox}{Characteristic polynomial}
$\lambda$ is an eigenvalue iff $A - \lambda I$ is singular, i.e.\
\[
p_{A}(\lambda) \;:=\; \det(A - \lambda I) \;=\; 0.
\]
$p_{A}$ is a polynomial of degree $n$ in $\lambda$.
\end{rulebox}

\begin{rulebox}{Eigenspace}
For an eigenvalue $\lambda$, the corresponding eigenspace is
$E_{\lambda} = \ker(A - \lambda I) = \{v : A v = \lambda v\}$.
\end{rulebox}

\begin{rulebox}{Trace and determinant identities}
For any $A \in \R^{n \times n}$ with eigenvalues
$\lambda_{1}, \dots, \lambda_{n}$ (with multiplicity):
\[
\mathrm{tr}(A) \;=\; \sum_{i} \lambda_{i}, \qquad
\det(A) \;=\; \prod_{i} \lambda_{i}.
\]
\end{rulebox}

\begin{rulebox}{Diagonalisation}
$A$ is \emph{diagonalisable} if it has $n$ linearly independent eigenvectors
$v_{1}, \dots, v_{n}$. Put them as columns of $P$ and the eigenvalues on the
diagonal of $D$; then
\[
A \;=\; P D P^{-1}, \qquad A^{k} \;=\; P D^{k} P^{-1}.
\]
Sufficient condition: $n$ distinct eigenvalues $\Rightarrow$ diagonalisable.
A symmetric real matrix is always diagonalisable (and via an orthogonal $P$,
with real eigenvalues, spectral theorem).
\end{rulebox}

\begin{rulebox}{Triangular shortcut}
For triangular $A$, the eigenvalues are exactly the diagonal entries.
\end{rulebox}

\section{Practice Problems}

\begin{problems}
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 2 & 0 \\ 0 & 3 \end{smallmatrix}\bigr)$.
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 4 & 1 \\ 0 & 3 \end{smallmatrix}\bigr)$.
  \item Find the eigenvalues and eigenvectors of $A = \bigl(\begin{smallmatrix} 2 & 1 \\ 1 & 2 \end{smallmatrix}\bigr)$.
  \item Find the eigenvalues and eigenvectors of $A = \bigl(\begin{smallmatrix} 3 & 1 \\ 0 & 3 \end{smallmatrix}\bigr)$. Is $A$ diagonalisable?
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 0 & 1 \\ 1 & 0 \end{smallmatrix}\bigr)$ and the corresponding eigenvectors.
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 0 & -1 \\ 1 & 0 \end{smallmatrix}\bigr)$ (real rotation). Real eigenvalues?
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 5 & 2 \\ 2 & 5 \end{smallmatrix}\bigr)$ and eigenvectors.
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 1 & 4 \\ 2 & 3 \end{smallmatrix}\bigr)$ and eigenvectors.
  \item Find the eigenvalues of $A = \begin{pmatrix} 2 & 0 & 0 \\ 1 & 2 & 0 \\ 3 & 1 & 4 \end{pmatrix}$.
  \item Find the eigenvalues of $A = \begin{pmatrix} 1 & 1 & 0 \\ 0 & 2 & 1 \\ 0 & 0 & 3 \end{pmatrix}$.
  \item Find the eigenvalues and eigenvectors of $A = \begin{pmatrix} 2 & 0 & 0 \\ 0 & 3 & 4 \\ 0 & 4 & 9 \end{pmatrix}$.
  \item Verify $\mathrm{tr}(A) = $ sum of eigenvalues and $\det A = $ product, for $A = \bigl(\begin{smallmatrix} 2 & 1 \\ 1 & 2 \end{smallmatrix}\bigr)$.
  \item Compute $A^{5}$ for $A = \bigl(\begin{smallmatrix} 2 & 0 \\ 0 & 3 \end{smallmatrix}\bigr)$.
  \item Diagonalise $A = \bigl(\begin{smallmatrix} 2 & 1 \\ 1 & 2 \end{smallmatrix}\bigr)$: find $P, D$ with $A = P D P^{-1}$.
  \item Using the diagonalisation, compute $A^{4}$ for $A = \bigl(\begin{smallmatrix} 2 & 1 \\ 1 & 2 \end{smallmatrix}\bigr)$.
  \item Show that if $v$ is an eigenvector of $A$ with eigenvalue $\lambda$, then $v$ is also an eigenvector of $A^{2}$ with eigenvalue $\lambda^{2}$.
  \item If $A$ is invertible and $\lambda$ is an eigenvalue of $A$, what is the corresponding eigenvalue of $A^{-1}$?
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 6 & -2 \\ 2 & 2 \end{smallmatrix}\bigr)$.
  \item Find the eigenvalues of $A = \begin{pmatrix} 1 & 0 & 1 \\ 0 & 1 & 0 \\ 1 & 0 & 1 \end{pmatrix}$ and an eigenvector for the largest one.
  \item For $A = \bigl(\begin{smallmatrix} 0.8 & 0.3 \\ 0.2 & 0.7 \end{smallmatrix}\bigr)$ (a stochastic matrix), find the eigenvalues.
  \item For the same $A$, find the eigenvector for $\lambda = 1$ (the stationary distribution).
  \item Show that $\lambda = 0$ is an eigenvalue of $A$ iff $A$ is singular.
  \item Find the eigenvalues of $A = \bigl(\begin{smallmatrix} 7 & 4 \\ -6 & -3 \end{smallmatrix}\bigr)$.
  \item Compute the characteristic polynomial of $A = \begin{pmatrix} 1 & 2 & 0 \\ 0 & 3 & 0 \\ 4 & 0 & 5 \end{pmatrix}$ and find its eigenvalues.
  \item For a $2 \times 2$ symmetric matrix $A$ with $\mathrm{tr}(A) = 5$ and $\det A = 6$, find its eigenvalues without computing $A$ explicitly.
\end{problems}

\section{Solutions}

\begin{solutions}
  \item Diagonal: $\lambda = 2, 3$.
  \item Upper-triangular: $\lambda = 4, 3$.
  \item $p_{A}(\lambda) = (2 - \lambda)^{2} - 1 = \lambda^{2} - 4 \lambda + 3 = (\lambda - 1)(\lambda - 3)$. Eigenvalues $\lambda = 1, 3$.
  For $\lambda = 1$: $A - I = \bigl(\begin{smallmatrix} 1 & 1 \\ 1 & 1 \end{smallmatrix}\bigr)$, eigenvector $(1, -1)^{T}$.
  For $\lambda = 3$: $A - 3 I = \bigl(\begin{smallmatrix} -1 & 1 \\ 1 & -1 \end{smallmatrix}\bigr)$, eigenvector $(1, 1)^{T}$.
  \item Upper-triangular: $\lambda = 3$ (double). $A - 3 I = \bigl(\begin{smallmatrix} 0 & 1 \\ 0 & 0 \end{smallmatrix}\bigr)$, rank $1$, eigenspace $1$-dimensional spanned by $(1, 0)^{T}$. Geometric multiplicity $1 <$ algebraic multiplicity $2$, so $A$ is \emph{not} diagonalisable.
  \item $p_{A}(\lambda) = \lambda^{2} - 1$, so $\lambda = \pm 1$. For $\lambda = 1$: eigenvector $(1, 1)^{T}$. For $\lambda = -1$: eigenvector $(1, -1)^{T}$.
  \item $p_{A}(\lambda) = \lambda^{2} + 1$, roots $\pm i$. No real eigenvalues.
  \item $p_{A}(\lambda) = (5 - \lambda)^{2} - 4 = \lambda^{2} - 10 \lambda + 21 = (\lambda - 3)(\lambda - 7)$. Eigenvalues $3, 7$. For $\lambda = 3$: $A - 3 I = \bigl(\begin{smallmatrix} 2 & 2 \\ 2 & 2 \end{smallmatrix}\bigr)$, eigenvector $(1, -1)^{T}$. For $\lambda = 7$: $(1, 1)^{T}$.
  \item $p_{A}(\lambda) = (1 - \lambda)(3 - \lambda) - 8 = \lambda^{2} - 4 \lambda - 5 = (\lambda - 5)(\lambda + 1)$. Eigenvalues $5, -1$.
  For $\lambda = 5$: $A - 5 I = \bigl(\begin{smallmatrix} -4 & 4 \\ 2 & -2 \end{smallmatrix}\bigr)$, eigenvector $(1, 1)^{T}$.
  For $\lambda = -1$: $A + I = \bigl(\begin{smallmatrix} 2 & 4 \\ 2 & 4 \end{smallmatrix}\bigr)$, eigenvector $(2, -1)^{T}$.
  \item Lower-triangular: $\lambda = 2, 2, 4$.
  \item Upper-triangular: $\lambda = 1, 2, 3$.
  \item Block diagonal: $\lambda = 2$ (from upper $1 \times 1$ block), and from the $2 \times 2$ block $\bigl(\begin{smallmatrix} 3 & 4 \\ 4 & 9 \end{smallmatrix}\bigr)$, $p(\lambda) = (3 - \lambda)(9 - \lambda) - 16 = \lambda^{2} - 12 \lambda + 11 = (\lambda - 1)(\lambda - 11)$. Eigenvalues $\lambda = 1, 2, 11$.
  Eigenvectors: for $\lambda = 2$: $e_{1} = (1, 0, 0)^{T}$. For $\lambda = 1$ (from $2 \times 2$ block $\bigl(\begin{smallmatrix} 2 & 4 \\ 4 & 8 \end{smallmatrix}\bigr)$): $(2, -1)^{T}$ in the lower block, so $(0, 2, -1)^{T}$. For $\lambda = 11$: $(3 - 11)v_{2} + 4 v_{3} = 0 \Rightarrow v_{3} = 2 v_{2}$, so $(0, 1, 2)^{T}$.
  \item Eigenvalues $\{1, 3\}$. Sum $= 4 = \mathrm{tr}(A) = 2 + 2$. Product $= 3 = \det A = 4 - 1$. Check.
  \item $A^{5} = \mathrm{diag}(2^{5}, 3^{5}) = \mathrm{diag}(32, 243)$.
  \item Eigenvalues $1, 3$ with eigenvectors $(1, -1)^{T}, (1, 1)^{T}$. So $P = \bigl(\begin{smallmatrix} 1 & 1 \\ -1 & 1 \end{smallmatrix}\bigr)$, $D = \mathrm{diag}(1, 3)$. $\det P = 2$, $P^{-1} = \tfrac{1}{2}\bigl(\begin{smallmatrix} 1 & -1 \\ 1 & 1 \end{smallmatrix}\bigr)$.
  \item $A^{4} = P D^{4} P^{-1} = P \mathrm{diag}(1, 81) P^{-1}$.
  \[
  P D^{4} = \begin{pmatrix} 1 & 81 \\ -1 & 81 \end{pmatrix}, \quad P D^{4} P^{-1} = \tfrac{1}{2}\begin{pmatrix} 1 & 81 \\ -1 & 81 \end{pmatrix} \begin{pmatrix} 1 & -1 \\ 1 & 1 \end{pmatrix} = \tfrac{1}{2}\begin{pmatrix} 82 & 80 \\ 80 & 82 \end{pmatrix} = \begin{pmatrix} 41 & 40 \\ 40 & 41 \end{pmatrix}.
  \]
  Sanity check via direct: $A^{2} = \bigl(\begin{smallmatrix} 5 & 4 \\ 4 & 5 \end{smallmatrix}\bigr)$, $A^{4} = (A^{2})^{2} = \bigl(\begin{smallmatrix} 25 + 16 & 20 + 20 \\ 20 + 20 & 16 + 25 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} 41 & 40 \\ 40 & 41 \end{smallmatrix}\bigr)$. Good.
  \item $A^{2} v = A(A v) = A(\lambda v) = \lambda A v = \lambda^{2} v$.
  \item $A v = \lambda v \Rightarrow v = A^{-1} \lambda v \Rightarrow A^{-1} v = \tfrac{1}{\lambda} v$. So eigenvalue is $1/\lambda$ (and $\lambda \neq 0$ since $A$ is invertible).
  \item $p_{A}(\lambda) = (6 - \lambda)(2 - \lambda) + 4 = \lambda^{2} - 8 \lambda + 16 = (\lambda - 4)^{2}$. Double eigenvalue $\lambda = 4$.
  \item Row $1 = $ Row $3$, so $\det A = 0$ and $\lambda = 0$ is an eigenvalue. The middle block gives $\lambda = 1$. For the remaining eigenvalue, $\mathrm{tr}(A) = 1 + 1 + 1 = 3$, so $\lambda_{1} + \lambda_{2} + \lambda_{3} = 0 + 1 + \lambda_{3} = 3 \Rightarrow \lambda_{3} = 2$.
  For $\lambda = 2$: $(A - 2 I)v = 0$ with $A - 2 I = \begin{pmatrix} -1 & 0 & 1 \\ 0 & -1 & 0 \\ 1 & 0 & -1 \end{pmatrix}$. Rows: $-v_{1} + v_{3} = 0$, $-v_{2} = 0$. So $v = (1, 0, 1)^{T}$.
  \item $p_{A}(\lambda) = (0.8 - \lambda)(0.7 - \lambda) - 0.06 = \lambda^{2} - 1.5 \lambda + 0.56 - 0.06 = \lambda^{2} - 1.5 \lambda + 0.5 = (\lambda - 1)(\lambda - 0.5)$. So $\lambda = 1, 0.5$.
  \item $A - I = \bigl(\begin{smallmatrix} -0.2 & 0.3 \\ 0.2 & -0.3 \end{smallmatrix}\bigr)$. Set $-0.2 v_{1} + 0.3 v_{2} = 0 \Rightarrow v_{1} = 1.5 v_{2}$. Choose $v_{2} = 2$: $v = (3, 2)^{T}$. Normalised to a probability vector: $(3/5, 2/5)^{T} = (0.6, 0.4)^{T}$.
  \item $\lambda = 0$ is an eigenvalue $\Leftrightarrow$ there exists $v \neq 0$ with $A v = 0$ $\Leftrightarrow$ $\ker A \neq \{0\}$ $\Leftrightarrow$ $A$ is singular.
  \item $p_{A}(\lambda) = (7 - \lambda)(-3 - \lambda) + 24 = \lambda^{2} - 4 \lambda + 3 = (\lambda - 1)(\lambda - 3)$. Eigenvalues $1, 3$.
  \item Expand $\det(A - \lambda I)$ along column 2: only the $(2, 2)$-entry is nonzero in that column (entries are $2, 3 - \lambda, 0$). Actually let's just expand directly.
  \[
  \det\!\begin{pmatrix} 1 - \lambda & 2 & 0 \\ 0 & 3 - \lambda & 0 \\ 4 & 0 & 5 - \lambda \end{pmatrix} = (3 - \lambda) \det \begin{pmatrix} 1 - \lambda & 0 \\ 4 & 5 - \lambda \end{pmatrix} = (3 - \lambda)(1 - \lambda)(5 - \lambda).
  \]
  (Cofactor along row 2 picks out only the $(2, 2)$ minor.) Eigenvalues $\lambda = 1, 3, 5$.
  \item $\lambda_{1} + \lambda_{2} = 5$, $\lambda_{1} \lambda_{2} = 6$. So $\lambda_{1}, \lambda_{2}$ are roots of $t^{2} - 5 t + 6 = (t - 2)(t - 3) = 0$. Eigenvalues $2, 3$.
\end{solutions}
