\chapter{Inverses and Linear Systems}

A square matrix $A \in \R^{n \times n}$ is \emph{invertible} (or
\emph{nonsingular}) if there exists a matrix $A^{-1}$ with
$A A^{-1} = A^{-1} A = I_{n}$. Equivalent statements: $\det A \neq 0$; the
columns of $A$ are linearly independent; the system $A x = b$ has a unique
solution for every $b$. The inverse is the workhorse for solving systems by
hand and theoretically (think $\hat{\beta} = (X^{T} X)^{-1} X^{T} y$ or
$x = (I - A)^{-1} d$ in input-output models).

\section{Core Rules}

\begin{rulebox}{$2 \times 2$ formula}
For $A = \bigl(\begin{smallmatrix} a & b \\ c & d \end{smallmatrix}\bigr)$
with $\det A = a d - b c \neq 0$,
\[
A^{-1} \;=\; \frac{1}{a d - b c} \begin{pmatrix} d & -b \\ -c & a \end{pmatrix}.
\]
(Swap the diagonal, negate the off-diagonal, divide by the determinant.)
\end{rulebox}

\begin{rulebox}{General formula (adjugate)}
For $A \in \R^{n \times n}$ invertible,
\[
A^{-1} \;=\; \frac{1}{\det A} \, \mathrm{adj}(A), \qquad
\mathrm{adj}(A)_{ij} = C_{ji} = (-1)^{i + j} M_{ji},
\]
where $C_{ji}$ are the cofactors of $A$. In practice for $n \ge 3$ one
usually inverts by Gauss-Jordan elimination on $[A \mid I]$ instead.
\end{rulebox}

\begin{rulebox}{Properties}
Whenever the inverses exist:
\begin{itemize}[leftmargin=1.6em,itemsep=0.3ex]
    \item $(A^{-1})^{-1} = A$.
    \item $(A B)^{-1} = B^{-1} A^{-1}$ \emph{(order reverses)}.
    \item $(A^{T})^{-1} = (A^{-1})^{T}$.
    \item $(\lambda A)^{-1} = \lambda^{-1} A^{-1}$ for $\lambda \neq 0$.
    \item $\det(A^{-1}) = 1 / \det A$.
    \item For a diagonal matrix $D = \mathrm{diag}(d_{1}, \dots, d_{n})$
          with all $d_{i} \neq 0$,
          $D^{-1} = \mathrm{diag}(1/d_{1}, \dots, 1/d_{n})$.
\end{itemize}
\end{rulebox}

\begin{rulebox}{Solving $A x = b$}
If $A$ is invertible, $x = A^{-1} b$ is the unique solution. In practice
Gauss elimination on $[A \mid b]$ is faster than computing $A^{-1}$ and
multiplying.
\end{rulebox}

\begin{rulebox}{Cramer's rule}
For $A x = b$ with $A \in \R^{n \times n}$ invertible,
\[
x_{i} \;=\; \frac{\det A_{i}}{\det A},
\]
where $A_{i}$ is $A$ with column $i$ replaced by $b$.
\end{rulebox}

\section{Practice Problems}

\begin{problems}
    \item Invert $A = \bigl(\begin{smallmatrix} 1 & 2 \\ 3 & 4 \end{smallmatrix}\bigr)$.
    \item Invert $A = \bigl(\begin{smallmatrix} 2 & 5 \\ 1 & 3 \end{smallmatrix}\bigr)$.
    \item Invert $A = \bigl(\begin{smallmatrix} 4 & 7 \\ 2 & 6 \end{smallmatrix}\bigr)$.
    \item For which $a$ is $A = \bigl(\begin{smallmatrix} a & 2 \\ 3 & a + 1 \end{smallmatrix}\bigr)$ invertible?
    \item Solve $\bigl(\begin{smallmatrix} 1 & 2 \\ 3 & 4 \end{smallmatrix}\bigr) x = \bigl(\begin{smallmatrix} 5 \\ 11 \end{smallmatrix}\bigr)$ using $A^{-1}$ (from problem 1).
    \item Solve $\bigl(\begin{smallmatrix} 2 & 5 \\ 1 & 3 \end{smallmatrix}\bigr) x = \bigl(\begin{smallmatrix} 1 \\ 0 \end{smallmatrix}\bigr)$ by Cramer's rule.
    \item Solve the system $\,2 x + 3 y = 7,\, x - y = 1\,$ using Cramer.
    \item Invert the diagonal matrix $D = \mathrm{diag}(2, -3, 5)$.
    \item Invert $A = \begin{pmatrix} 1 & 0 & 0 \\ 2 & 1 & 0 \\ 3 & 4 & 1 \end{pmatrix}$ (lower-triangular with $1$'s on the diagonal).
    \item Invert $A = \begin{pmatrix} 1 & 2 & 3 \\ 0 & 1 & 4 \\ 0 & 0 & 1 \end{pmatrix}$.
    \item Invert $A = \begin{pmatrix} 1 & 2 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 2 \end{pmatrix}$.
    \item Invert $A = \begin{pmatrix} 1 & 1 & 1 \\ 0 & 1 & 1 \\ 0 & 0 & 1 \end{pmatrix}$.
    \item Solve $\begin{pmatrix} 1 & 1 & 1 \\ 0 & 1 & 1 \\ 0 & 0 & 1 \end{pmatrix} x = \begin{pmatrix} 6 \\ 5 \\ 3 \end{pmatrix}$ by back-substitution.
    \item Solve $\begin{pmatrix} 2 & 1 & 0 \\ 1 & 3 & 1 \\ 0 & 1 & 2 \end{pmatrix} x = \begin{pmatrix} 3 \\ 5 \\ 3 \end{pmatrix}$ by Gauss elimination.
    \item If $A = \bigl(\begin{smallmatrix} 1 & 2 \\ 3 & 4 \end{smallmatrix}\bigr)$ and $B = \bigl(\begin{smallmatrix} 2 & 0 \\ 0 & 5 \end{smallmatrix}\bigr)$, compute $(A B)^{-1}$ two ways: directly and via $B^{-1} A^{-1}$.
    \item For $A \in \R^{3 \times 3}$ with $\det A = 2$, what is $\det(A^{-1})$? $\det(2 A^{-1})$?
    \item Show that if $A$ is invertible and symmetric, then so is $A^{-1}$.
    \item For $A$ as in problem 1, verify that $A A^{-1} = I_{2}$ by direct multiplication.
    \item Solve $A x = b$ with $A = \bigl(\begin{smallmatrix} 1 & 2 \\ 3 & 7 \end{smallmatrix}\bigr)$ and $b = \bigl(\begin{smallmatrix} 4 \\ 9 \end{smallmatrix}\bigr)$.
    \item For which $\lambda$ is $A - \lambda I_{2}$ not invertible, where $A = \bigl(\begin{smallmatrix} 2 & 1 \\ 0 & 3 \end{smallmatrix}\bigr)$?
    \item Compute the inverse of a rotation matrix $R(\theta) = \bigl(\begin{smallmatrix} \cos \theta & -\sin \theta \\ \sin \theta & \cos \theta \end{smallmatrix}\bigr)$.
    \item Find $X$ from $X A = B$ with $A = \bigl(\begin{smallmatrix} 1 & 2 \\ 3 & 4 \end{smallmatrix}\bigr)$ and $B = \bigl(\begin{smallmatrix} 2 & 0 \\ -1 & 1 \end{smallmatrix}\bigr)$.
    \item Compute $A^{-1}$ for $A = \begin{pmatrix} 2 & 0 & 1 \\ 0 & 1 & 0 \\ 1 & 0 & 1 \end{pmatrix}$.
    \item Solve $\begin{pmatrix} 2 & 0 & 1 \\ 0 & 1 & 0 \\ 1 & 0 & 1 \end{pmatrix} x = \begin{pmatrix} 3 \\ 2 \\ 2 \end{pmatrix}$.
    \item True or false: if $A \in \R^{3 \times 3}$ has two equal columns then $A$ is not invertible. Justify.
\end{problems}

\section{Solutions}

\begin{solutions}
    \item $\det A = 1 \cdot 4 - 2 \cdot 3 = -2$. So $A^{-1} = -\tfrac{1}{2} \bigl(\begin{smallmatrix} 4 & -2 \\ -3 & 1 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} -2 & 1 \\ 3/2 & -1/2 \end{smallmatrix}\bigr)$.
    \item $\det A = 6 - 5 = 1$. $A^{-1} = \bigl(\begin{smallmatrix} 3 & -5 \\ -1 & 2 \end{smallmatrix}\bigr)$.
    \item $\det A = 24 - 14 = 10$. $A^{-1} = \tfrac{1}{10} \bigl(\begin{smallmatrix} 6 & -7 \\ -2 & 4 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} 3/5 & -7/10 \\ -1/5 & 2/5 \end{smallmatrix}\bigr)$.
    \item $\det A = a(a + 1) - 6 = a^{2} + a - 6 = (a - 2)(a + 3)$. Invertible iff $a \neq 2, -3$.
    \item $x = A^{-1} b = \bigl(\begin{smallmatrix} -2 & 1 \\ 3/2 & -1/2 \end{smallmatrix}\bigr) \bigl(\begin{smallmatrix} 5 \\ 11 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} -10 + 11 \\ 15/2 - 11/2 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} 1 \\ 2 \end{smallmatrix}\bigr)$.
    \item $\det A = 1$. $A_{1} = \bigl(\begin{smallmatrix} 1 & 5 \\ 0 & 3 \end{smallmatrix}\bigr)$, $\det A_{1} = 3$. $A_{2} = \bigl(\begin{smallmatrix} 2 & 1 \\ 1 & 0 \end{smallmatrix}\bigr)$, $\det A_{2} = -1$. So $x_{1} = 3,\, x_{2} = -1$.
    \item $A = \bigl(\begin{smallmatrix} 2 & 3 \\ 1 & -1 \end{smallmatrix}\bigr)$, $\det A = -2 - 3 = -5$. $A_{x} = \bigl(\begin{smallmatrix} 7 & 3 \\ 1 & -1 \end{smallmatrix}\bigr)$, $\det = -7 - 3 = -10$. $A_{y} = \bigl(\begin{smallmatrix} 2 & 7 \\ 1 & 1 \end{smallmatrix}\bigr)$, $\det = 2 - 7 = -5$. So $x = -10/-5 = 2$, $y = -5/-5 = 1$.
    \item $D^{-1} = \mathrm{diag}(1/2, -1/3, 1/5)$.
    \item Row-reduce $[A \mid I]$. $R_{2} \to R_{2} - 2 R_{1}$, $R_{3} \to R_{3} - 3 R_{1}$ gives $\bigl(\begin{smallmatrix} 1 & 0 & 0 \\ 0 & 1 & 0 \\ 0 & 4 & 1 \end{smallmatrix}\big| \begin{smallmatrix} 1 & 0 & 0 \\ -2 & 1 & 0 \\ -3 & 0 & 1 \end{smallmatrix}\bigr)$. Then $R_{3} \to R_{3} - 4 R_{2}$:
    \[
    A^{-1} = \begin{pmatrix} 1 & 0 & 0 \\ -2 & 1 & 0 \\ 5 & -4 & 1 \end{pmatrix}.
    \]
    Check: $A A^{-1}$: row 3 of $A$ is $(3, 4, 1)$; dotted with cols of $A^{-1}$ gives $(3 - 8 + 5,\, 0 + 4 - 4,\, 0 + 0 + 1) = (0, 0, 1)$. Good.
    \item Use the unit upper-triangular structure. The inverse is unit upper-triangular. Row-reduce $[A \mid I]$: clear column 3 first using $R_{1} \to R_{1} - 3 R_{3}$, $R_{2} \to R_{2} - 4 R_{3}$:
    \[
    \begin{pmatrix} 1 & 2 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1 \end{pmatrix} \Big| \begin{pmatrix} 1 & 0 & -3 \\ 0 & 1 & -4 \\ 0 & 0 & 1 \end{pmatrix}.
    \]
    Now $R_{1} \to R_{1} - 2 R_{2}$:
    \[
    A^{-1} = \begin{pmatrix} 1 & -2 & 5 \\ 0 & 1 & -4 \\ 0 & 0 & 1 \end{pmatrix}.
    \]
    Check the $(1, 3)$ entry of $A A^{-1}$: $1 \cdot 5 + 2 \cdot (-4) + 3 \cdot 1 = 5 - 8 + 3 = 0$. Good.
    \item Block-by-block: the upper-left $2 \times 2$ block $\bigl(\begin{smallmatrix} 1 & 2 \\ 0 & 1 \end{smallmatrix}\bigr)$ has inverse $\bigl(\begin{smallmatrix} 1 & -2 \\ 0 & 1 \end{smallmatrix}\bigr)$, and the $(3,3)$ block is $1/2$.
    \[
    A^{-1} = \begin{pmatrix} 1 & -2 & 0 \\ 0 & 1 & 0 \\ 0 & 0 & 1/2 \end{pmatrix}.
    \]
    \item Row-reduce $[A \mid I]$: $R_{1} \to R_{1} - R_{2}$, $R_{2} \to R_{2} - R_{3}$:
    \[
    A^{-1} = \begin{pmatrix} 1 & -1 & 0 \\ 0 & 1 & -1 \\ 0 & 0 & 1 \end{pmatrix}.
    \]
    Verify $A A^{-1}$ has the right diagonal of $1$'s and zeros off-diagonal.
    \item Back-substitute: $x_{3} = 3$; $x_{2} + x_{3} = 5 \Rightarrow x_{2} = 2$; $x_{1} + x_{2} + x_{3} = 6 \Rightarrow x_{1} = 1$.
    \item Augmented matrix: $\bigl(\begin{smallmatrix} 2 & 1 & 0 & 3 \\ 1 & 3 & 1 & 5 \\ 0 & 1 & 2 & 3 \end{smallmatrix}\bigr)$. $R_{2} \to R_{2} - \tfrac{1}{2} R_{1}$: $(0, 5/2, 1, 7/2)$. $R_{3}$ unchanged. Then $R_{3} \to R_{3} - \tfrac{2}{5} R_{2}$: $(0, 0, 2 - 2/5, 3 - 7/5) = (0, 0, 8/5, 8/5)$. So $x_{3} = 1$. Back-sub: $\tfrac{5}{2} x_{2} + 1 = 7/2 \Rightarrow x_{2} = 1$. $2 x_{1} + 1 = 3 \Rightarrow x_{1} = 1$. So $x = (1, 1, 1)^{T}$.
    \item Direct: $A B = \bigl(\begin{smallmatrix} 2 & 10 \\ 6 & 20 \end{smallmatrix}\bigr)$, $\det = 40 - 60 = -20$, so $(A B)^{-1} = -\tfrac{1}{20}\bigl(\begin{smallmatrix} 20 & -10 \\ -6 & 2 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} -1 & 1/2 \\ 3/10 & -1/10 \end{smallmatrix}\bigr)$.
    Via $B^{-1} A^{-1}$: $B^{-1} = \mathrm{diag}(1/2, 1/5)$, $A^{-1} = \bigl(\begin{smallmatrix} -2 & 1 \\ 3/2 & -1/2 \end{smallmatrix}\bigr)$.
    \[
    B^{-1} A^{-1} = \begin{pmatrix} 1/2 & 0 \\ 0 & 1/5 \end{pmatrix} \begin{pmatrix} -2 & 1 \\ 3/2 & -1/2 \end{pmatrix} = \begin{pmatrix} -1 & 1/2 \\ 3/10 & -1/10 \end{pmatrix}.
    \]
    Match.
    \item $\det(A^{-1}) = 1/2$. $\det(2 A^{-1}) = 2^{3} \cdot \det(A^{-1}) = 8 \cdot \tfrac{1}{2} = 4$.
    \item From $A A^{-1} = I$, transpose: $(A^{-1})^{T} A^{T} = I$. If $A^{T} = A$, this gives $(A^{-1})^{T} A = I$, so $(A^{-1})^{T} = A^{-1}$, i.e.\ $A^{-1}$ is symmetric.
    \item $A A^{-1} = \bigl(\begin{smallmatrix} 1 & 2 \\ 3 & 4 \end{smallmatrix}\bigr) \bigl(\begin{smallmatrix} -2 & 1 \\ 3/2 & -1/2 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} -2 + 3 & 1 - 1 \\ -6 + 6 & 3 - 2 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} 1 & 0 \\ 0 & 1 \end{smallmatrix}\bigr)$.
    \item $\det A = 7 - 6 = 1$, $A^{-1} = \bigl(\begin{smallmatrix} 7 & -2 \\ -3 & 1 \end{smallmatrix}\bigr)$. $x = A^{-1} b = \bigl(\begin{smallmatrix} 28 - 18 \\ -12 + 9 \end{smallmatrix}\bigr) = \bigl(\begin{smallmatrix} 10 \\ -3 \end{smallmatrix}\bigr)$.
    \item $\det(A - \lambda I) = (2 - \lambda)(3 - \lambda) - 0 = 0 \Rightarrow \lambda = 2$ or $\lambda = 3$.
    \item $\det R(\theta) = \cos^{2}\theta + \sin^{2}\theta = 1$, so $R^{-1} = \bigl(\begin{smallmatrix} \cos \theta & \sin \theta \\ -\sin \theta & \cos \theta \end{smallmatrix}\bigr) = R(-\theta)$.
    \item $X = B A^{-1}$. From problem 1, $A^{-1} = \bigl(\begin{smallmatrix} -2 & 1 \\ 3/2 & -1/2 \end{smallmatrix}\bigr)$. So
    \[
    X = \begin{pmatrix} 2 & 0 \\ -1 & 1 \end{pmatrix} \begin{pmatrix} -2 & 1 \\ 3/2 & -1/2 \end{pmatrix} = \begin{pmatrix} -4 & 2 \\ 2 + 3/2 & -1 - 1/2 \end{pmatrix} = \begin{pmatrix} -4 & 2 \\ 7/2 & -3/2 \end{pmatrix}.
    \]
    \item $\det A$ along row 2: $1 \cdot \det\bigl(\begin{smallmatrix} 2 & 1 \\ 1 & 1 \end{smallmatrix}\bigr) = 1$. Row-reduce $[A \mid I]$. Swap to expose the structure, or compute cofactors directly. By the adjugate, cofactor matrix is
    \[
    C = \begin{pmatrix} 1 & 0 & -1 \\ 0 & 1 & 0 \\ -1 & 0 & 2 \end{pmatrix},
    \]
    so $A^{-1} = \mathrm{adj}(A) = C^{T} = \begin{pmatrix} 1 & 0 & -1 \\ 0 & 1 & 0 \\ -1 & 0 & 2 \end{pmatrix}$ (symmetric since $A$ is). Verify: $A A^{-1}$ row 1: $(2 \cdot 1 + 0 + 1 \cdot (-1),\, 0,\, 2 \cdot (-1) + 0 + 1 \cdot 2) = (1, 0, 0)$. Good.
    \item $x = A^{-1} b = \begin{pmatrix} 1 & 0 & -1 \\ 0 & 1 & 0 \\ -1 & 0 & 2 \end{pmatrix} \begin{pmatrix} 3 \\ 2 \\ 2 \end{pmatrix} = \begin{pmatrix} 1 \\ 2 \\ 1 \end{pmatrix}$.
    \item True. Two equal columns imply linear dependence of columns, so $\det A = 0$ (or equivalently rank $< 3$), so $A$ is not invertible.
\end{solutions}
