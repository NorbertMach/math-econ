\section{Core Rules}

\begin{rulebox}{What the notation abbreviates}
\[
  \sum_{i=1}^{n}a_{i}=a_{1}+a_{2}+\cdots+a_{n} .
\]
$i$ is a \textbf{dummy index}: it exists only inside the sum, and renaming
it changes nothing ($\sum_{i}a_{i}=\sum_{j}a_{j}$). A quantity carrying no
index --- $c$, $n$, $\bar a$ --- is constant with respect to $i$ and can be
moved outside.
\end{rulebox}

\begin{rulebox}{Linearity, and the constant sum}
\[
  \sum_{i=1}^{n}(a_{i}+b_{i})=\sum a_{i}+\sum b_{i},\qquad
  \sum_{i=1}^{n}c\,a_{i}=c\sum a_{i},\qquad
  \sum_{i=1}^{n}c=nc .
\]
The third is the one that is forgotten: summing a constant $n$ times gives
$nc$, not $c$. Together these handle every expression of the form
$\sum(\alpha a_{i}+\beta)$.
\end{rulebox}

\begin{rulebox}{The closed forms}
\[
  \sum_{k=1}^{n}k=\frac{n(n+1)}{2},\qquad
  \sum_{k=1}^{n}k^{2}=\frac{n(n+1)(2n+1)}{6},\qquad
  \sum_{k=1}^{n}k^{3}=\Bigl(\frac{n(n+1)}{2}\Bigr)^{2},
\]
\[
  \sum_{k=0}^{n}c\,r^{k}=c\,\frac{1-r^{n+1}}{1-r}\quad(r\neq1) .
\]
The geometric one has $n+1$ terms because it starts at $k=0$; that
off-by-one is worth checking every time.
\end{rulebox}

\begin{rulebox}{What is \emph{not} linear}
\[
  \sum a_{i}^{2}\neq\Bigl(\sum a_{i}\Bigr)^{2},\qquad
  \sum a_{i}b_{i}\neq\Bigl(\sum a_{i}\Bigr)\Bigl(\sum b_{i}\Bigr),\qquad
  \sum\frac{1}{a_{i}}\neq\frac{1}{\sum a_{i}} .
\]
Sums pass through addition and scalar multiples, and through nothing else.
The gap between $\sum a_{i}^{2}$ and $(\sum a_{i})^{2}$ is, up to a factor,
the variance --- so it is not a small technicality but the subject matter of
statistics.
\end{rulebox}

\begin{rulebox}{Two structural identities}
With $\bar a=\frac1n\sum a_{i}$, deviations from the mean sum to zero:
\[
  \sum_{i=1}^{n}(a_{i}-\bar a)=\sum a_{i}-n\bar a=n\bar a-n\bar a=0 .
\]
And a telescoping sum collapses to its ends:
\[
  \sum_{i=1}^{n}(a_{i+1}-a_{i})=a_{n+1}-a_{1} .
\]
\end{rulebox}

\section{Worked Example}

\begin{worked}{Prove $\sum_{i=1}^{n}(a_{i}-\bar a)^{2}=\sum_{i=1}^{n}a_{i}^{2}-n\bar a^{2}$, then verify it on $a=(3,-1,2,4)$}
\wstep{Step 1. Expand the square inside the sum.}
\[
  (a_{i}-\bar a)^{2}=a_{i}^{2}-2\bar a\,a_{i}+\bar a^{2} .
\]
Expand \emph{before} summing: the summation sign passes over sums of terms,
so once the bracket is expanded the sum splits.

\wstep{Step 2. Split by linearity.}
\[
  \sum_{i=1}^{n}(a_{i}-\bar a)^{2}
  =\sum_{i=1}^{n}a_{i}^{2}-2\bar a\sum_{i=1}^{n}a_{i}+\sum_{i=1}^{n}\bar a^{2} .
\]
\wtrap{$\bar a$ carries no index $i$, so it is a constant for this sum and
comes out of the middle term. This is the step students hesitate over ---
$\bar a$ is built from all the $a_{i}$, so it \emph{feels} like it varies.
It does not: it is one number, computed once.}

\wstep{Step 3. Deal with the two remaining sums.}
By definition $\sum a_{i}=n\bar a$, and the last sum adds the constant
$\bar a^{2}$ exactly $n$ times:
\[
  =\sum a_{i}^{2}-2\bar a\,(n\bar a)+n\bar a^{2}
  =\sum a_{i}^{2}-2n\bar a^{2}+n\bar a^{2}
  =\boxed{\sum_{i=1}^{n}a_{i}^{2}-n\bar a^{2}} .
\]
\wtrap{$\sum_{i=1}^{n}\bar a^{2}=n\bar a^{2}$, not $\bar a^{2}$. Constants do
not ``pass through'' a sum untouched --- they get counted $n$ times.}

\wstep{Step 4. Verify numerically, which is how you catch a slip.}
With $a=(3,-1,2,4)$ and $n=4$: $\sum a_{i}=8$, so $\bar a=2$.

\emph{Right-hand side:}
$\sum a_{i}^{2}=9+1+4+16=30$, and $n\bar a^{2}=4\cdot4=16$, giving
$30-16=14$.

\emph{Left-hand side:} the deviations are $1,-3,0,2$, and
$1+9+0+4=14$. \checkmark

\wstep{Step 5. Note what the two sides are for.}
The left side says what the quantity \emph{means} --- total squared
dispersion about the mean. The right side is how it is \emph{computed}: one
pass over the data accumulating $\sum a_{i}$ and $\sum a_{i}^{2}$, with no
need to know the mean in advance. Divided by $n$ it is the variance, and the
same identity in two variables produces the covariance, and with it the
OLS slope formula of \chref{matrices}{10}.
\wtrap{Check the deviations sum to zero --- here $1-3+0+2=0$ --- before
squaring them. If they do not, the mean is wrong and everything after it
is too.}
\end{worked}

  \item Expand: $\displaystyle\sum_{k=1}^{n}k$
  \item Expand: $\displaystyle\sum_{i=2}^{4}a_{i}^{2}$
  \item Expand: $\displaystyle\sum_{j=0}^{3}2^{j}$
  \item Compute: $\displaystyle\sum_{k=1}^{5}k$
  \item Compute: $\displaystyle\sum_{k=1}^{4}k^{2}$
  \item Compute: $\displaystyle\sum_{k=1}^{3}k^{3}$
  \item Geometric series formula: $\displaystyle\sum_{k=0}^{n}c\,r^{k}$ for
        $r\neq 1$.
  \item Simplify: $\displaystyle\sum_{i=1}^{n}(a_{i}+c)-nc$
  \item Simplify: $\displaystyle\sum_{i=1}^{n}(a_{i}-\bar a)$
  \item Simplify: $\displaystyle\frac{1}{n}\sum_{i=1}^{n}a_{i}^{2}-\bar
        a^{2}$ (name the quantity)
  \item Prove: $\displaystyle\sum_{i=1}^{n}(a_{i}-\bar a)^{2}
        =\sum_{i=1}^{n}a_{i}^{2}-n\bar a^{2}$
  \item Simplify: $\displaystyle\sum_{i=1}^{n}c\,a_{i}b_{i}$
  \item Index shift: rewrite $\displaystyle\sum_{i=1}^{n}a_{i+1}$ in terms
        of $a_{j}$.
  \item Telescoping: $\displaystyle\sum_{i=1}^{n}(a_{i+1}-a_{i})$
  \item Simplify: $\displaystyle\sum_{i=1}^{n}i(i+1)$
  \item Evaluate: $\bar a$
  \item Evaluate: $\bar b$
  \item Evaluate: $\displaystyle\sum_{i=1}^{4}a_{i}^{2}$
  \item Evaluate: $\displaystyle\left(\sum_{i=1}^{4}a_{i}\right)^{2}$
  \item Evaluate: $\displaystyle\sum_{i=1}^{4}a_{i}b_{i}$
  \item Evaluate: $\displaystyle\sum_{i=1}^{4}(a_{i}-\bar a)(b_{i}-\bar b)$
  \item Evaluate: $\displaystyle\sum_{i=1}^{4}(a_{i}-\bar a)^{2}$
  \item Evaluate: $\displaystyle\sum_{i=1}^{4}(3a_{i}+2)$

  \item Evaluate: $\displaystyle\sum_{i=1}^{3}(-1)^{i}\,i$
  \item Evaluate: $\displaystyle\sum_{k=0}^{4}\left(\frac{1}{2}\right)^{k}$
  \item Present value: $\displaystyle\sum_{t=1}^{3}\frac{100}{(1.1)^{t}}$

        $\displaystyle\sum_{i=1}^{10}(2i-1)$

  \item True or false (with one-line justification):
        $\displaystyle\left|\sum a_{i}\right|\le\sum|a_{i}|$
\end{problems}

\section{Solutions}

\begin{solutions}
  \item $1+2+\cdots+n$
  \item $a_{2}^{2}+a_{3}^{2}+a_{4}^{2}$
  \item $1+2+4+8=15$
  \item $\dfrac{5\cdot 6}{2}=15$
  \item $\dfrac{4\cdot 5\cdot 9}{6}=30$
  \item $\left(\dfrac{3\cdot 4}{2}\right)^{2}=36$
  \item $c\cdot\dfrac{r^{n+1}-1}{r-1}$
  \item $\displaystyle\sum a_{i}$
  \item $0$
  \item Population variance of $\{a_{i}\}$.
  \item Expand: $\sum(a_{i}^{2}-2\bar a a_{i}+\bar a^{2})
        =\sum a_{i}^{2}-2\bar a\,n\bar a+n\bar a^{2}
        =\sum a_{i}^{2}-n\bar a^{2}$.
  \item $c\displaystyle\sum a_{i}b_{i}$
  \item $\displaystyle\sum_{j=2}^{n+1}a_{j}$
  \item $a_{n+1}-a_{1}$
  \item $\dfrac{n(n+1)(n+2)}{3}$
  \item $\bar a=\dfrac{8}{4}=2$
  \item $\bar b=\dfrac{4}{4}=1$
  \item $9+1+4+16=30$
  \item $8^{2}=64$
  \item $3-2-2+8=7$
  \item $(1)(0)+(-3)(1)+(0)(-2)+(2)(1)=-1$
  \item $1+9+0+4=14$
  \item $3\cdot 8+2\cdot 4=32$
  \item $16+1+1+36=54$ (consistent: $30+2(7)+10=54$)
  \item $-1+2-3=-2$
  \item $\dfrac{1-(1/2)^{5}}{1-1/2}=\dfrac{31}{16}$
  \item $\dfrac{100}{1.1}+\dfrac{100}{1.21}+\dfrac{100}{1.331}\approx
        248.69$
  \item $10^{2}=100$
  \item $5\cdot 7=35$
  \item True. This is the triangle inequality applied iteratively to
        $|a_{1}+\cdots+a_{n}|$.
\end{solutions}
